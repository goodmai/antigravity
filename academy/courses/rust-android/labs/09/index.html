Lab 09 — Foreground Service + Rust engine

Задача: long-running Rust-таска (например, индексирование локальных файлов или peer-to-peer sync), которая переживает kill системы и видна пользователю как уведомление.

Kotlin: сервис-обёртка

class SyncService : Service() {
    private var engine: SyncEngine? = null

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        startForeground(1, buildNotification("Syncing…"))
        engine = SyncEngine().also { it.start() }
        return START_STICKY
    }

    override fun onDestroy() {
        engine?.stop()
        engine = null
    }
    override fun onBind(intent: Intent?) = null
}

Rust: бесконечный tick через tokio

use std::sync::Arc;
use tokio::{runtime::Runtime, sync::Notify, time::{interval, Duration}};

#[derive(uniffi::Object)]
pub struct SyncEngine {
    rt: Runtime,
    stop: Arc<Notify>,
}

#[uniffi::export]
impl SyncEngine {
    #[uniffi::constructor]
    pub fn new() -> Arc<Self> {
        Arc::new(Self {
            rt: Runtime::new().unwrap(),
            stop: Arc::new(Notify::new()),
        })
    }

    pub fn start(self: Arc<Self>) {
        let stop = self.stop.clone();
        self.rt.spawn(async move {
            let mut tick = interval(Duration::from_secs(15));
            loop {
                tokio::select! {
                    _ = tick.tick() => { do_one_round().await; }
                    _ = stop.notified() => break,
                }
            }
        });
    }

    pub fn stop(&self) { self.stop.notify_waiters(); }
}

async fn do_one_round() {
    // здесь: HTTP-sync, IPFS-publish, что угодно.
    log::info!("tick");
}

Каналы: Rust → Kotlin (events)

Чтобы UI получал прогресс, опишите Kotlin-интерфейс Listener и прокиньте его в Rust. UniFFI поддерживает callback-интерфейсы: 

#[uniffi::export(with_foreign)]
pub trait Listener: Send + Sync {
    fn on_progress(&self, done: u32, total: u32);
}

#[uniffi::export]
impl SyncEngine {
    pub fn set_listener(&self, l: Box<dyn Listener>) { /* store and call */ }
}
