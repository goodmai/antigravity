Глобальный Runtime

tokio-runtime создаём один раз при загрузке .so через JNI_OnLoad и кладём в once_cell::sync::OnceCell. Иначе на каждый вызов будем плодить новые рантаймы — это сожжёт батарею. 

use once_cell::sync::OnceCell;
use tokio::runtime::Runtime;

static RT: OnceCell<Runtime> = OnceCell::new();

fn rt() -> &'static Runtime {
    RT.get_or_init(|| Runtime::new().unwrap())
}

Async-функция и JNI-обёртка

#[derive(serde::Deserialize)]
struct Ticker { #[serde(rename = "USD")] usd: f64 }

async fn btc_price() -> reqwest::Result<f64> {
    let t: Ticker = reqwest::Client::new()
        .get("https://api.coindesk.com/v1/bpi/currentprice/USD.json")
        .send().await?
        .error_for_status()?
        .json::<serde_json::Value>().await?
        ["bpi"]["USD"]["rate_float"]
        .as_f64()
        .map(|usd| Ticker { usd })
        .unwrap_or(Ticker { usd: 0.0 });
    Ok(t.usd)
}

#[no_mangle]
pub extern "system" fn Java_io_daskibo_btc_NativeKt_btcPrice(
    mut env: jni::JNIEnv,
    _: jni::objects::JClass,
) -> jni::sys::jdouble {
    rt().block_on(btc_price()).unwrap_or(-1.0)
}

Kotlin: запускаем в coroutine

lifecycleScope.launch(Dispatchers.IO) {
    val usd = Native.btcPrice()    // блокирует Rust-поток, но не Kotlin'овский UI
    withContext(Dispatchers.Main) {
        priceView.text = "BTC = $${"%.2f".format(usd)}"
    }
}
