UniFFI: идиоматичные биндинги Rust↔Kotlin

Писать JNI вручную — это правильно для понимания, но скучно в продакшене. UniFFI (от Mozilla, использовался в Firefox Sync, Mozilla VPN, Glean) автоматически генерирует Kotlin / Swift / Python / Ruby обёртки из вашего Rust API. С 2024 года поддерживается proc-macro-режим: больше не нужны UDL-файлы.

1. Описание API через атрибуты

// src/lib.rs
use uniffi::Object;

#[derive(uniffi::Error, Debug, thiserror::Error)]
pub enum NoteError {
    #[error("note not found")]
    NotFound,
}

#[derive(Object)]
pub struct NoteStore {
    inner: std::sync::Mutex<Vec<String>>,
}

#[uniffi::export]
impl NoteStore {
    #[uniffi::constructor]
    pub fn new() -> std::sync::Arc<Self> {
        std::sync::Arc::new(Self { inner: Default::default() })
    }

    pub fn add(&self, body: String) {
        self.inner.lock().unwrap().push(body);
    }

    pub fn get(&self, idx: u32) -> Result<String, NoteError> {
        self.inner.lock().unwrap().get(idx as usize).cloned().ok_or(NoteError::NotFound)
    }

    pub fn count(&self) -> u32 {
        self.inner.lock().unwrap().len() as u32
    }
}

uniffi::setup_scaffolding!();

2. Сборка биндингов

# build.rs
fn main() { uniffi::generate_scaffolding("src/note.udl").ok(); }

# затем:
cargo build --release --target aarch64-linux-android
uniffi-bindgen generate --library target/aarch64-linux-android/release/libnote.so \
    --language kotlin --out-dir bindings/

В bindings/ появится Kotlin-файл с такими же типами, как в Rust: 

// bindings/io/daskibo/note/note.kt   (сгенерировано)
class NoteStore {
    fun add(body: String)
    @Throws(NoteException::class)
    fun `get`(idx: UInt): String
    fun count(): UInt
}

3. Использование из Kotlin

val store = NoteStore()
store.add("first note")
store.add("second note")
val n = store.`get`(0u)  // "first note"
val total = store.count() // 2
