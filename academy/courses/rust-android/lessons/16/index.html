Логирование и пропуск ошибок через FFI

На Android println! не виден в Logcat — stdout редиректится в /dev/null. Используйте крейт android_logger: он пишет через __android_log_print и интегрируется с фасадом log.

Инициализация логгера в JNI_OnLoad

use jni::{JavaVM, sys::{jint, JNI_VERSION_1_6}};
use std::ffi::c_void;

#[no_mangle]
pub extern "system" fn JNI_OnLoad(_vm: JavaVM, _: *mut c_void) -> jint {
    android_logger::init_once(
        android_logger::Config::default()
            .with_max_level(log::LevelFilter::Info)
            .with_tag("daskibo"),
    );
    log::info!("Rust core загружен");
    JNI_VERSION_1_6
}

Panic не должен переходить границу FFI

panic! через ABI — это undefined behavior. Защититесь std::panic::catch_unwind и конвертируйте панику в ошибку Java-уровня: 

use std::panic::{catch_unwind, AssertUnwindSafe};

#[no_mangle]
pub extern "system" fn Java_io_daskibo_NativeKt_divide(
    mut env: jni::JNIEnv,
    _: jni::objects::JClass,
    a: jni::sys::jint,
    b: jni::sys::jint,
) -> jni::sys::jint {
    let res = catch_unwind(AssertUnwindSafe(|| {
        if b == 0 { panic!("div-by-zero"); }
        a / b
    }));
    match res {
        Ok(v) => v,
        Err(_) => {
            let _ = env.throw_new("java/lang/ArithmeticException", "rust panic");
            0
        }
    }
}

Задание: в песочнице добавьте кейс divide(i32::MIN, -1) — это переполнение, его тоже надо ловить. Подсказка: checked_div из стандартной библиотеки.
