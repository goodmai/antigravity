Lab 01 · Развёртывание первого ERC-20

В этом лабораторном задании вы пройдёте полный цикл от установки инструментов до верификации живого контракта в тестовой сети. Вы напишете минимальный токен на базе OpenZeppelin, покроете его Foundry-тестами, задеплоите на Sepolia и прочитаете состояние контракта через viem. По итогу у вас будет задеплоенный ERC-20 с адресом на Sepolia Etherscan — это первый «живой» артефакт в вашем Web3-портфолио.

Что вы сделаете:
  • Установите Foundry и инициализируете проект с зависимостью OpenZeppelin.
  • Напишете минимальный ERC-20 с функцией mint.
  • Покроете контракт Foundry-тестами (deploy, totalSupply, transfer, события).
  • Задеплоите контракт на Sepolia через forge script --broadcast.
  • Верифицируете исходный код на Sepolia Etherscan.
  • Прочитаете состояние через viem publicClient.readContract.
  • Отправите перевод через viem walletClient.writeContract и проверите событие.
Что сдать: ссылку на верифицированный контракт на Sepolia Etherscan и скриншот успешного теста forge test -vv.

Задание 1

Установите Foundry и инициализируйте новый проект. Добавьте OpenZeppelin Contracts как git-субмодуль через forge install. Убедитесь, что файл foundry.toml содержит корректный маппинг для импортов. 

# Установка Foundry (выполняется один раз)
curl -L https://foundry.paradigm.xyz | bash
foundryup

# Создание проекта
forge init my-token
cd my-token

# Добавление OpenZeppelin
forge install OpenZeppelin/openzeppelin-contracts --no-commit

# Проверка версии
forge --version
# forge 0.3.x (...)

После установки добавьте в foundry.toml ремапинг: 

[profile.default]
src     = "src"
out     = "out"
libs    = ["lib"]
remappings = [
    "@openzeppelin/contracts/=lib/openzeppelin-contracts/contracts/"
]

Критерий приёмки: команда forge build завершается без ошибок; папка lib/openzeppelin-contracts/ существует.

Задание 2

Напишите минимальный ERC-20 контракт в файле src/MyToken.sol. Контракт должен наследовать OpenZeppelin ERC20, иметь конструктор с именем и символом токена, и публичную функцию mint(address, uint256), доступную только owner'у. 

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

import "@openzeppelin/contracts/token/ERC20/ERC20.sol";
import "@openzeppelin/contracts/access/Ownable.sol";

contract MyToken is ERC20, Ownable {
    constructor(
        string memory name_,
        string memory symbol_,
        address initialOwner
    )
        ERC20(name_, symbol_)
        Ownable(initialOwner)
    {}

    /// @notice Чеканит токены на адрес `to`. Только owner.
    function mint(address to, uint256 amount) external onlyOwner {
        _mint(to, amount);
    }
}

Критерий приёмки: forge build компилируется без предупреждений; в артефакте out/MyToken.sol/MyToken.json присутствует ABI с функциями mint, transfer, balanceOf.

Задание 3

Напишите Foundry-тест в файле test/MyToken.t.sol. Тест должен: задеплоить контракт; проверить начальный totalSupply; вызвать mint и проверить баланс; сделать transfer и проверить балансы обеих сторон; убедиться, что события Transfer эмитируются корректно через vm.expectEmit. 

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

import "forge-std/Test.sol";
import "../src/MyToken.sol";

contract MyTokenTest is Test {
    MyToken public token;
    address public owner  = address(0xA11CE);
    address public alice  = address(0xA11C3);
    address public bob    = address(0xB0B);

    function setUp() public {
        vm.startPrank(owner);
        token = new MyToken("My Token", "MTK", owner);
        vm.stopPrank();
    }

    function test_InitialSupplyIsZero() public view {
        assertEq(token.totalSupply(), 0);
    }

    function test_MintIncreasesSupply() public {
        vm.prank(owner);
        token.mint(alice, 1000e18);
        assertEq(token.totalSupply(), 1000e18);
        assertEq(token.balanceOf(alice), 1000e18);
    }

    function test_TransferMovesBalance() public {
        vm.prank(owner);
        token.mint(alice, 500e18);

        vm.prank(alice);
        token.transfer(bob, 200e18);

        assertEq(token.balanceOf(alice), 300e18);
        assertEq(token.balanceOf(bob),   200e18);
    }

    function test_TransferEmitsEvent() public {
        vm.prank(owner);
        token.mint(alice, 100e18);

        // vm.expectEmit(checkTopic1, checkTopic2, checkTopic3, checkData, emitter)
        vm.expectEmit(true, true, false, true, address(token));
        emit IERC20.Transfer(alice, bob, 50e18);

        vm.prank(alice);
        token.transfer(bob, 50e18);
    }

    function test_MintRevertsForNonOwner() public {
        vm.prank(alice);
        vm.expectRevert();
        token.mint(alice, 1e18);
    }
}

Критерий приёмки: forge test -vv показывает 5 зелёных тестов без единого FAIL.

Задание 4

Напишите Foundry-скрипт деплоя в script/Deploy.s.sol и задеплойте контракт на Sepolia через forge script --broadcast. Для этого нужен RPC-URL от Alchemy/Infura и приватный ключ тестового кошелька (никогда не используйте mainnet-ключ в скриптах). 

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

import "forge-std/Script.sol";
import "../src/MyToken.sol";

contract DeployMyToken is Script {
    function run() external {
        uint256 deployerKey = vm.envUint("PRIVATE_KEY");
        address deployer    = vm.addr(deployerKey);

        vm.startBroadcast(deployerKey);
        MyToken token = new MyToken("My Token", "MTK", deployer);
        // Чеканим 1 000 000 токенов для деплоера
        token.mint(deployer, 1_000_000e18);
        vm.stopBroadcast();

        console.log("MyToken deployed at:", address(token));
    }
}

# .env (не коммитить!)
PRIVATE_KEY=0xac0974bec39a17e36ba4a6b4d238ff944bacb478cbed5efcae784d7bf4f2ff80
SEPOLIA_RPC_URL=https://eth-sepolia.g.alchemy.com/v2/YOUR_API_KEY
ETHERSCAN_API_KEY=YOUR_ETHERSCAN_KEY

# Загружаем переменные и запускаем деплой
source .env
forge script script/Deploy.s.sol:DeployMyToken \
  --rpc-url $SEPOLIA_RPC_URL \
  --broadcast \
  --verify \
  -vvvv

Критерий приёмки: в выводе присутствует строка MyToken deployed at: 0x…; транзакция деплоя видна на sepolia.etherscan.io.

Задание 5

Если верификация через --verify не сработала автоматически (бывает при congestion Etherscan), выполните её вручную через forge verify-contract. 

# Узнаём аргументы конструктора в ABI-кодировании
cast abi-encode "constructor(string,string,address)" \
  "My Token" "MTK" 0xYOUR_DEPLOYER_ADDRESS

# Верификация вручную
forge verify-contract \
  0xYOUR_CONTRACT_ADDRESS \
  src/MyToken.sol:MyToken \
  --chain-id 11155111 \
  --etherscan-api-key $ETHERSCAN_API_KEY \
  --constructor-args $(cast abi-encode "constructor(string,string,address)" \
      "My Token" "MTK" 0xYOUR_DEPLOYER_ADDRESS)

# Проверка статуса верификации
forge verify-check \
  GUID_FROM_PREVIOUS_COMMAND \
  --chain-id 11155111 \
  --etherscan-api-key $ETHERSCAN_API_KEY

Критерий приёмки: на странице контракта в Sepolia Etherscan появилась вкладка «Contract» со статусом «Verified» и читаемым исходным кодом Solidity.

Задание 6

Прочитайте состояние задеплоенного контракта через viem publicClient.readContract. Проверьте: name(), symbol(), totalSupply(), balanceOf(deployer). 

import { createPublicClient, http, formatUnits } from "viem";
import { sepolia } from "viem/chains";

const CONTRACT = "0xYOUR_CONTRACT_ADDRESS";
const DEPLOYER  = "0xYOUR_DEPLOYER_ADDRESS";

const ABI = [
  { name: "name",        type: "function", inputs: [], outputs: [{ type: "string" }],  stateMutability: "view" },
  { name: "symbol",      type: "function", inputs: [], outputs: [{ type: "string" }],  stateMutability: "view" },
  { name: "totalSupply", type: "function", inputs: [], outputs: [{ type: "uint256" }], stateMutability: "view" },
  { name: "balanceOf",   type: "function",
    inputs: [{ name: "account", type: "address" }],
    outputs: [{ type: "uint256" }],
    stateMutability: "view"
  },
] as const;

const client = createPublicClient({
  chain:     sepolia,
  transport: http("https://eth-sepolia.g.alchemy.com/v2/YOUR_API_KEY"),
});

const [name, symbol, totalSupply, balance] = await Promise.all([
  client.readContract({ address: CONTRACT, abi: ABI, functionName: "name" }),
  client.readContract({ address: CONTRACT, abi: ABI, functionName: "symbol" }),
  client.readContract({ address: CONTRACT, abi: ABI, functionName: "totalSupply" }),
  client.readContract({ address: CONTRACT, abi: ABI, functionName: "balanceOf", args: [DEPLOYER] }),
]);

console.log(`${name} (${symbol})`);
console.log(`totalSupply: ${formatUnits(totalSupply, 18)} MTK`);
console.log(`deployer balance: ${formatUnits(balance, 18)} MTK`);

Критерий приёмки: скрипт выводит корректное имя, символ и баланс деплоера = 1 000 000 MTK.

Задание 7

Отправьте перевод токенов через viem walletClient.writeContract. После отправки дождитесь подтверждения транзакции, затем получите receipt и проверьте, что в логах присутствует событие Transfer с правильными аргументами from, to, value. 

import {
  createPublicClient, createWalletClient,
  http, custom, parseUnits, decodeEventLog
} from "viem";
import { sepolia } from "viem/chains";
import { privateKeyToAccount } from "viem/accounts";

const CONTRACT = "0xYOUR_CONTRACT_ADDRESS";
const RECIPIENT = "0xRECIPIENT_ADDRESS";

const ABI = [
  {
    name: "transfer",
    type: "function",
    inputs: [
      { name: "to",     type: "address" },
      { name: "amount", type: "uint256" },
    ],
    outputs: [{ type: "bool" }],
    stateMutability: "nonpayable",
  },
  {
    name: "Transfer",
    type: "event",
    inputs: [
      { name: "from",  type: "address", indexed: true  },
      { name: "to",    type: "address", indexed: true  },
      { name: "value", type: "uint256", indexed: false },
    ],
  },
] as const;

const account = privateKeyToAccount("0xYOUR_PRIVATE_KEY");

const publicClient = createPublicClient({
  chain: sepolia,
  transport: http("https://eth-sepolia.g.alchemy.com/v2/YOUR_API_KEY"),
});

const walletClient = createWalletClient({
  account,
  chain:     sepolia,
  transport: http("https://eth-sepolia.g.alchemy.com/v2/YOUR_API_KEY"),
});

// Отправляем 100 MTK
const hash = await walletClient.writeContract({
  address:      CONTRACT,
  abi:          ABI,
  functionName: "transfer",
  args:         [RECIPIENT, parseUnits("100", 18)],
});

console.log("tx hash:", hash);

// Ждём подтверждения
const receipt = await publicClient.waitForTransactionReceipt({ hash });
console.log("status:", receipt.status); // 'success'

// Декодируем событие Transfer
for (const log of receipt.logs) {
  try {
    const decoded = decodeEventLog({ abi: ABI, ...log });
    if (decoded.eventName === "Transfer") {
      console.log("Transfer event:", decoded.args);
      // { from: '0x...', to: '0x...', value: 100000000000000000000n }
    }
  } catch { /* не наш log */ }
}

Критерий приёмки: транзакция подтверждена (status: success); декодированное событие содержит корректные from, to и value = 100e18; баланс получателя вырос на 100 MTK.
