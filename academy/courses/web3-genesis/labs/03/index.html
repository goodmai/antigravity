Lab 03 · Ownable & AccessControl

Управление правами доступа — один из наиболее критичных аспектов безопасности смарт-контрактов. В этой лабораторной работе вы начнёте с простого Ownable, перейдёте к гибкому AccessControl с несколькими ролями, добавите механизм паузы и в финале реализуете упрощённый 2-of-3 мультисиг-голосования на базе ролей. Все сценарии покрываются Foundry-тестами, финальный контракт деплоится на Sepolia.

Что вы сделаете:
  • Напишете и протестируете контракт с Ownable: mint только owner.
  • Проверите передачу и отречение от владения с проверкой событий.
  • Перейдёте на AccessControl: MINTER_ROLE, PAUSER_ROLE, DEFAULT_ADMIN_ROLE.
  • Напишете тест: admin выдаёт MINTER_ROLE → minter может mint; не-minter — revert.
  • Реализуете паузу/снятие с паузы через Pausable.
  • Реализуете 2-of-3 голосование через AccessControl (3 guardian'а, пауза при 2 голосах).
  • Задеплоите контракт на Sepolia и продемонстрируете управление ролями через cast.
Что сдать: адрес контракта на Sepolia + транскрипт cast-команд с назначением роли и вызовом защищённой функции.

Задание 1

Напишите src/OwnableToken.sol с модификатором onlyOwner на функции mint. Напишите тест, проверяющий, что не-owner не может mint, и что передача owner'шипа работает корректно. 

// src/OwnableToken.sol
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

import "@openzeppelin/contracts/token/ERC20/ERC20.sol";
import "@openzeppelin/contracts/access/Ownable.sol";

contract OwnableToken is ERC20, Ownable {
    constructor(address initialOwner)
        ERC20("Ownable Token", "OWT")
        Ownable(initialOwner)
    {}

    function mint(address to, uint256 amount) external onlyOwner {
        _mint(to, amount);
    }
}

// test/OwnableTokenTest.t.sol
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

import "forge-std/Test.sol";
import "../src/OwnableToken.sol";

contract OwnableTokenTest is Test {
    OwnableToken token;
    address owner   = address(0x1);
    address newOwner = address(0x2);
    address hacker  = address(0x3);

    function setUp() public {
        vm.prank(owner);
        token = new OwnableToken(owner);
    }

    function test_OnlyOwnerCanMint() public {
        vm.prank(owner);
        token.mint(owner, 1000e18);
        assertEq(token.balanceOf(owner), 1000e18);
    }

    function test_NonOwnerMintReverts() public {
        vm.prank(hacker);
        vm.expectRevert();
        token.mint(hacker, 1000e18);
    }

    function test_TransferOwnership() public {
        vm.prank(owner);
        token.transferOwnership(newOwner);
        assertEq(token.owner(), newOwner);

        // Новый owner может mint
        vm.prank(newOwner);
        token.mint(newOwner, 500e18);
        assertEq(token.balanceOf(newOwner), 500e18);

        // Старый owner — больше нет
        vm.prank(owner);
        vm.expectRevert();
        token.mint(owner, 1e18);
    }
}

Критерий приёмки: все тесты зелёные; вы понимаете разницу между transferOwnership и Ownable2Step.transferOwnership.

Задание 2

Проверьте в тесте события при передаче и отречении от владения. Используйте vm.expectEmit для обоих сценариев. 

function test_TransferOwnershipEmitsEvent() public {
    vm.expectEmit(true, true, false, false, address(token));
    emit Ownable.OwnershipTransferred(owner, newOwner);

    vm.prank(owner);
    token.transferOwnership(newOwner);
}

function test_RenounceOwnership() public {
    vm.expectEmit(true, true, false, false, address(token));
    emit Ownable.OwnershipTransferred(owner, address(0));

    vm.prank(owner);
    token.renounceOwnership();

    assertEq(token.owner(), address(0));

    // После renounce никто не может mint
    vm.prank(owner);
    vm.expectRevert();
    token.mint(owner, 1e18);
}
Внимание. renounceOwnership() необратима — после неё контракт навсегда теряет owner'а. Для продакшн-контрактов с критичными admin-функциями рекомендуется Ownable2Step, который требует явного принятия владения новым адресом.

Критерий приёмки: оба теста проходят; в комментарии написано, почему renounceOwnership опасна для контрактов с ценными admin-функциями.

Задание 3

Перепишите контракт с использованием AccessControl. Определите три роли: MINTER_ROLE, PAUSER_ROLE, DEFAULT_ADMIN_ROLE. Admin должен иметь возможность выдавать и отзывать любую роль. 

// src/AccessToken.sol
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

import "@openzeppelin/contracts/token/ERC20/ERC20.sol";
import "@openzeppelin/contracts/access/AccessControl.sol";
import "@openzeppelin/contracts/utils/Pausable.sol";

contract AccessToken is ERC20, AccessControl, Pausable {
    bytes32 public constant MINTER_ROLE = keccak256("MINTER_ROLE");
    bytes32 public constant PAUSER_ROLE = keccak256("PAUSER_ROLE");

    constructor(address admin) ERC20("Access Token", "ACT") {
        // DEFAULT_ADMIN_ROLE выдаётся admin'у
        _grantRole(DEFAULT_ADMIN_ROLE, admin);
    }

    function mint(address to, uint256 amount)
        external
        onlyRole(MINTER_ROLE)
        whenNotPaused
    {
        _mint(to, amount);
    }

    function pause()   external onlyRole(PAUSER_ROLE) { _pause(); }
    function unpause() external onlyRole(PAUSER_ROLE) { _unpause(); }

    // Переопределяем _update для поддержки Pausable
    function _update(address from, address to, uint256 value)
        internal
        override
        whenNotPaused
    {
        super._update(from, to, value);
    }
}

Критерий приёмки: компилируется без ошибок; функция supportsInterface возвращает true для IAccessControl и IERC20.

Задание 4

Напишите тесты для AccessControl: admin выдаёт MINTER_ROLE → minter mintит; не-minter ревертится; admin отзывает роль → minter больше не может mintить. 

// test/AccessTokenTest.t.sol

pragma solidity ^0.8.24;

import "forge-std/Test.sol";
import "../src/AccessToken.sol";

contract AccessTokenTest is Test {

    address admin  = address(0x1);
    address minter = address(0x2);
    address pauser = address(0x3);
    address alice  = address(0x4);

    function setUp() public {
        vm.prank(admin);
        token = new AccessToken(admin);
    }

    function test_AdminCanGrantMinterRole() public {
        vm.prank(admin);
        token.grantRole(token.MINTER_ROLE(), minter);
        assertTrue(token.hasRole(token.MINTER_ROLE(), minter));
    }

    function test_MinterCanMint() public {
        vm.prank(admin);
        token.grantRole(token.MINTER_ROLE(), minter);

        vm.prank(minter);
        token.mint(alice, 1000e18);
        assertEq(token.balanceOf(alice), 1000e18);
    }

    function test_NonMinterMintReverts() public {
        vm.prank(alice);
        vm.expectRevert();
        token.mint(alice, 1e18);

        token.grantRole(token.MINTER_ROLE(), minter);

        vm.stopPrank();

        vm.prank(minter);
        vm.expectRevert();
        token.mint(alice, 1e18);
    }

    function test_NonAdminCannotGrantRole() public {
        vm.prank(alice);
        vm.expectRevert();
        token.grantRole(token.MINTER_ROLE(), alice);
    }
}

Задание 5

Реализуйте тесты для механизма паузы. Убедитесь, что при паузе функции mint и transfer ревертятся, а после снятия паузы — работают. Проверьте события Paused и Unpaused. 

function test_PauserCanPauseAndUnpause() public {
    vm.prank(admin);
    token.grantRole(token.PAUSER_ROLE(), pauser);
    token.grantRole(token.MINTER_ROLE(), minter);

    // Mint до паузы
    vm.prank(minter);
    token.mint(alice, 100e18);

    // Пауза
    vm.expectEmit(false, false, false, true, address(token));
    emit Pausable.Paused(pauser);

    vm.prank(pauser);
    token.pause();
    assertTrue(token.paused());

    // Mint при паузе — revert
    vm.prank(minter);
    vm.expectRevert();
    token.mint(alice, 1e18);

    // Transfer при паузе — revert
    vm.prank(alice);
    vm.expectRevert();
    token.transfer(address(0x5), 10e18);

    // Снятие паузы
    vm.prank(pauser);
    token.unpause();

    // Mint снова работает
    vm.prank(minter);
    token.mint(alice, 50e18);
    assertEq(token.balanceOf(alice), 150e18);
}

Критерий приёмки: тест проходит; события Paused/Unpaused правильно декодируются.

Задание 6

Реализуйте 2-of-3 мультисиг-голосование для паузы: три guardian'а имеют GUARDIAN_ROLE; пауза активируется только при голосовании 2 из 3. Напишите соответствующие тесты. 

// Добавьте в AccessToken.sol или создайте отдельный контракт:

bytes32 public constant GUARDIAN_ROLE = keccak256("GUARDIAN_ROLE");
uint256 public pauseVotes;
mapping(address => bool) public hasPausedVoted;

function votePause() external onlyRole(GUARDIAN_ROLE) {
    require(!hasPausedVoted[msg.sender], "already voted");
    hasPausedVoted[msg.sender] = true;
    pauseVotes++;

    if (pauseVotes >= 2) {
        _pause();
    }
}

function resetPauseVotes() external onlyRole(DEFAULT_ADMIN_ROLE) {
    // Сброс голосов при снятии паузы
    pauseVotes = 0;
    // Сбросить mapping нельзя дёшево — используем nonce-паттерн в продакшене
}

function test_TwoOfThreeGuardiansCanPause() public {
    address g1 = address(0xG1);
    address g2 = address(0xG2);
    address g3 = address(0xG3);

    vm.startPrank(admin);
    token.grantRole(token.GUARDIAN_ROLE(), g1);
    token.grantRole(token.GUARDIAN_ROLE(), g2);
    token.grantRole(token.GUARDIAN_ROLE(), g3);
    vm.stopPrank();

    // Один голос — пауза не активируется
    vm.prank(g1);
    token.votePause();
    assertFalse(token.paused());

    // Второй голос — пауза!
    vm.prank(g2);
    token.votePause();
    assertTrue(token.paused());

    // Третий guardian не может голосовать дважды
    vm.prank(g1);
    vm.expectRevert("already voted");
    token.votePause();
}

Критерий приёмки: тесты проходят; в комментарии описан способ реализации сброса mapping через nonce-паттерн для продакшена.

Задание 7

Задеплойте контракт на Sepolia и продемонстрируйте управление ролями через cast send. Используйте cast call для чтения ролей и cast для кодирования аргументов. 

# Переменные
export CONTRACT=0xYOUR_CONTRACT
export ADMIN_KEY=0xYOUR_ADMIN_PRIVATE_KEY
export MINTER=0xMINTER_ADDRESS
export RPC=https://eth-sepolia.g.alchemy.com/v2/YOUR_KEY

# Вычислить байты32 роли
MINTER_ROLE=$(cast keccak "MINTER_ROLE")
echo "MINTER_ROLE: $MINTER_ROLE"

# Проверить роли до выдачи
cast call $CONTRACT "hasRole(bytes32,address)" $MINTER_ROLE $MINTER --rpc-url $RPC

# Выдать MINTER_ROLE
cast send $CONTRACT \
  "grantRole(bytes32,address)" \
  $MINTER_ROLE $MINTER \
  --private-key $ADMIN_KEY \
  --rpc-url $RPC

# Проверить роль после выдачи
cast call $CONTRACT "hasRole(bytes32,address)" $MINTER_ROLE $MINTER --rpc-url $RPC
# => 0x0000...0001 (true)

# Вызвать mint от имени minter
cast send $CONTRACT \
  "mint(address,uint256)" \
  $MINTER 1000000000000000000000 \
  --private-key $MINTER_KEY \
  --rpc-url $RPC

# Проверить баланс
cast call $CONTRACT "balanceOf(address)" $MINTER --rpc-url $RPC | cast to-dec

Критерий приёмки: транзакции подтверждены на Sepolia Etherscan; скриншот или транскрипт показывает успешную выдачу роли и вызов mint.
