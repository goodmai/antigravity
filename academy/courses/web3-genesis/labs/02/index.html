Lab 02 · ERC-20 Allowance & Permit

Большинство DeFi-протоколов не работают с прямыми переводами — они используют механизм approve / transferFrom. В этой лабораторной работе вы исследуете стандартные allowances, разберёте anti-pattern бесконечного апрува, реализуете EIP-2612 permit() для gasless-подписей и встроите всё это в симулированный DEX. В итоге вы поймёте, почему современные протоколы переходят на Permit2 и как создать полноценный gasless-flow.

Что вы сделаете:
  • Реализуете и протестируете стандартный approve / transferFrom.
  • Проверите поведение MAX_UINT256-approve в Foundry-тесте.
  • Добавите EIP-2612 permit() через OpenZeppelin ERC20Permit.
  • Подпишете EIP-712 typed data через viem signTypedData.
  • Реализуете gasless transferFrom: пользователь подписывает, релеер отправляет.
  • Напишете фаззинг-тест для permit-подписи в Foundry.
  • Встроите approve + transferFrom в мок-DEX.
Что сдать: репозиторий с тестами (все зелёные), скриншот gasless-транзакции на Sepolia с разными адресами подписанта и отправителя.

Задание 1

Напишите контракт src/AllowanceToken.sol, унаследованный от ERC20. Добавьте тесты, проверяющие: нулевой allowance по умолчанию; что после approve allowance равен заданной сумме; что transferFrom уменьшает allowance; что повторный transferFrom на сумму сверх allowance ревертится. 

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

import "@openzeppelin/contracts/token/ERC20/ERC20.sol";

contract AllowanceToken is ERC20 {
    constructor() ERC20("Allowance Token", "ALW") {
        _mint(msg.sender, 1_000_000e18);
    }
}

// test/AllowanceTest.t.sol
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

import "forge-std/Test.sol";
import "../src/AllowanceToken.sol";

contract AllowanceTest is Test {
    AllowanceToken token;
    address owner = address(1);
    address spender = address(2);
    address recipient = address(3);

    function setUp() public {
        vm.prank(owner);
        token = new AllowanceToken();
    }

    function test_DefaultAllowanceIsZero() public view {
        assertEq(token.allowance(owner, spender), 0);
    }

    function test_ApproveSetAllowance() public {
        vm.prank(owner);
        token.approve(spender, 500e18);
        assertEq(token.allowance(owner, spender), 500e18);
    }

    function test_TransferFromDecreasesAllowance() public {
        vm.prank(owner);
        token.approve(spender, 500e18);

        vm.prank(spender);
        token.transferFrom(owner, recipient, 200e18);

        assertEq(token.allowance(owner, spender), 300e18);
    }

    function test_TransferFromRevertsOverAllowance() public {
        vm.prank(owner);
        token.approve(spender, 100e18);

        vm.prank(spender);
        vm.expectRevert();
        token.transferFrom(owner, recipient, 101e18);
    }
}

Критерий приёмки: все четыре теста проходят; добавьте тест на попытку transferFrom без approve — должен ревертиться.

Задание 2

Исследуйте «unlimited approve» anti-pattern: approve type(uint256).max, затем убедитесь, что после transferFrom allowance не уменьшился. Напишите тест, фиксирующий это поведение OpenZeppelin ERC20. 

function test_UnlimitedApproveDoesNotDecrease() public {
    uint256 MAX = type(uint256).max;

    vm.prank(owner);
    token.approve(spender, MAX);
    assertEq(token.allowance(owner, spender), MAX);

    vm.prank(spender);
    token.transferFrom(owner, recipient, 999e18);

    // Ключевая проверка: allowance остался MAX
    assertEq(token.allowance(owner, spender), MAX);
}

function test_UnlimitedApproveIsRisky() public {
    // Документируем риск: если spender скомпрометирован —
    // он может вывести ВСЕ токены owner'а
    uint256 MAX = type(uint256).max;
    uint256 ownerBalance = token.balanceOf(owner);

    vm.prank(owner);
    token.approve(spender, MAX);

    // Злоумышленник (spender) выводит всё
    vm.prank(spender);
    token.transferFrom(owner, spender, ownerBalance);

    assertEq(token.balanceOf(owner), 0);
    assertEq(token.balanceOf(spender), ownerBalance);
}
Почему это опасно. Большинство DEX-фронтендов по умолчанию предлагают «infinite approve» ради удобства (меньше транзакций). Но если контракт взломан или окажется вредоносным — он заберёт весь баланс. Всегда предпочитайте точный approve или используйте ERC-20 Permit (задание 3).

Критерий приёмки: оба теста проходят; в комментарии к коду объяснён риск unlimited approve.

Задание 3

Обновите контракт до ERC20Permit из OpenZeppelin и разберите устройство domain separator. Напишите тест, который проверяет, что DOMAIN_SEPARATOR() привязан к chainId и адресу контракта.

// test/PermitTest.t.sol
function test_DomainSeparatorBindsToChainAndAddress() public view {
    bytes32 expected = keccak256(abi.encode(
        keccak256("EIP712Domain(string name,string version,uint256 chainId,address verifyingContract)"),
        keccak256(bytes("Permit Token")),
        keccak256(bytes("1")),
        block.chainid,
        address(token)
    ));
    assertEq(token.DOMAIN_SEPARATOR(), expected);
}

Задание 4

Подпишите EIP-712 typed data для permit через viem signTypedData и вызовите permit() на контракте. Реализуйте весь flow: формирование структуры, подпись, вызов on-chain. 

import {
  createWalletClient, createPublicClient,
  http, custom, parseUnits
} from "viem";
import { sepolia } from "viem/chains";
import { privateKeyToAccount } from "viem/accounts";

const CONTRACT  = "0xYOUR_PERMIT_TOKEN";
const SPENDER   = "0xSPENDER_ADDRESS";
const account   = privateKeyToAccount("0xOWNER_PRIVATE_KEY");

const publicClient = createPublicClient({ chain: sepolia, transport: http() });

// Получаем текущий nonce для permit
const nonce = await publicClient.readContract({
  address:      CONTRACT,
  abi:          [{ name: "nonces", type: "function",
                   inputs: [{ type: "address" }], outputs: [{ type: "uint256" }],
                   stateMutability: "view" }],
  functionName: "nonces",
  args:         [account.address],
});

const deadline = BigInt(Math.floor(Date.now() / 1000) + 3600); // +1 час

// EIP-712 структура для permit
const walletClient = createWalletClient({
  account,
  chain:     sepolia,
  transport: custom(window.ethereum), // или http() если скрипт
});

const signature = await walletClient.signTypedData({
  domain: {
    name:              "Permit Token",
    version:           "1",
    chainId:           sepolia.id,
    verifyingContract: CONTRACT,
  },
  types: {
    Permit: [
      { name: "owner",    type: "address" },
      { name: "spender",  type: "address" },
      { name: "value",    type: "uint256" },
      { name: "nonce",    type: "uint256" },
      { name: "deadline", type: "uint256" },
    ],
  },
  primaryType: "Permit",
  message: {
    owner:    account.address,
    spender:  SPENDER,
    value:    parseUnits("1000", 18),
    nonce,
    deadline,
  },
});

// Разбиваем подпись на v, r, s
const { v, r, s } = parseSignature(signature); // из viem

console.log("Подпись готова:", { v, r, s, deadline });
// Теперь любой может вызвать permit() с этими параметрами

Критерий приёмки: подпись сформирована без ошибок; в логах распечатаны корректные значения v (27 или 28), r и s (по 32 байта каждый).

Задание 5

Реализуйте gasless transferFrom flow: пользователь подписывает permit офчейн, релеер (другой EOA) вызывает permit() + transferFrom() в одной транзакции, оплачивая газ сам. Напишите Foundry-тест, симулирующий этот сценарий. 

// test/GaslessTransferTest.t.sol
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

import "forge-std/Test.sol";
import "../src/PermitToken.sol";

contract GaslessTransferTest is Test {
    PermitToken token;

    // Фиксированный тестовый приватный ключ (только для тестов!)
    uint256 ownerKey  = 0xac0974bec39a17e36ba4a6b4d238ff944bacb478cbed5efcae784d7bf4f2ff80;
    address owner;
    address relayer   = address(0xRELA);
    address recipient = address(0xBEEF);

    function setUp() public {
        owner = vm.addr(ownerKey);
        vm.prank(owner);
        token = new PermitToken();
    }

    function test_GaslessTransferViaPermit() public {
        uint256 amount   = 500e18;
        uint256 deadline = block.timestamp + 1 hours;
        uint256 nonce    = token.nonces(owner);

        // 1. Формируем хэш permit-сообщения
        bytes32 structHash = keccak256(abi.encode(
            keccak256("Permit(address owner,address spender,uint256 value,uint256 nonce,uint256 deadline)"),
            owner,
            relayer,   // spender = relayer
            amount,
            nonce,
            deadline
        ));

        bytes32 digest = keccak256(abi.encodePacked(
            "\x19\x01",
            token.DOMAIN_SEPARATOR(),
            structHash
        ));

        // 2. Подписываем (офчейн, owner не платит газ)
        (uint8 v, bytes32 r, bytes32 s) = vm.sign(ownerKey, digest);

        // 3. Релеер вызывает permit + transferFrom (платит газ)
        vm.startPrank(relayer);
        token.permit(owner, relayer, amount, deadline, v, r, s);
        token.transferFrom(owner, recipient, amount);
        vm.stopPrank();

        assertEq(token.balanceOf(recipient), amount);
        assertEq(token.allowance(owner, relayer), 0); // использован полностью
    }
}

Критерий приёмки: тест проходит; в комментариях объяснено, кто платит газ и почему это называется «gasless» с точки зрения пользователя.

Задание 6

Напишите Foundry-фаззинг-тест для permit-подписи. Тест должен случайным образом генерировать параметры и проверять, что невалидная подпись всегда ревертится, а валидная всегда принимается. 

// Добавьте в PermitTest.t.sol
function testFuzz_InvalidSignatureReverts(
    uint256 wrongKey,
    uint256 amount,
    uint256 deadline
) public {
    // Ограничиваем входные данные
    vm.assume(wrongKey != ownerKey && wrongKey != 0);
    vm.assume(amount > 0 && amount <= token.balanceOf(owner));
    vm.assume(deadline > block.timestamp);

    uint256 nonce = token.nonces(owner);

    bytes32 structHash = keccak256(abi.encode(
        keccak256("Permit(address owner,address spender,uint256 value,uint256 nonce,uint256 deadline)"),
        owner,
        address(this),
        amount,
        nonce,
        deadline
    ));

    bytes32 digest = keccak256(abi.encodePacked(
        "\x19\x01",
        token.DOMAIN_SEPARATOR(),
        structHash
    ));

    // Подписываем НЕВЕРНЫМ ключом
    (uint8 v, bytes32 r, bytes32 s) = vm.sign(wrongKey, digest);

    // Должен ревертиться
    vm.expectRevert();
    token.permit(owner, address(this), amount, deadline, v, r, s);
}

# Запуск фаззинга
forge test --match-test testFuzz_InvalidSignatureReverts --fuzz-runs 1000 -v

Критерий приёмки: фаззинг прогоняет 1000 итераций без единого counterexample.

Задание 7

Напишите мок-контракт DEX и проверьте полный flow: пользователь апрувит токены DEX'у, DEX вызывает transferFrom. Затем повторите то же самое через permit-flow (без предварительного approve). 

// src/MockDex.sol
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

import "@openzeppelin/contracts/token/ERC20/IERC20.sol";
import "@openzeppelin/contracts/token/ERC20/extensions/IERC20Permit.sol";

contract MockDex {
    event Swapped(address indexed user, address token, uint256 amountIn);

    /// @notice Классический flow: пользователь сначала вызвал approve
    function swapExact(address token, uint256 amountIn) external {
        IERC20(token).transferFrom(msg.sender, address(this), amountIn);
        emit Swapped(msg.sender, token, amountIn);
        // ... логика обмена
    }

    /// @notice Gasless flow: подпись передаётся вместе со swap
    function swapExactWithPermit(
        address token,
        uint256 amountIn,
        uint256 deadline,
        uint8   v,
        bytes32 r,
        bytes32 s
    ) external {
        // Один вызов вместо двух транзакций от пользователя
        IERC20Permit(token).permit(msg.sender, address(this), amountIn, deadline, v, r, s);
        IERC20(token).transferFrom(msg.sender, address(this), amountIn);
        emit Swapped(msg.sender, token, amountIn);
    }
}

// test/MockDexTest.t.sol — напишите тесты для обоих методов:
// 1. test_SwapExactWithApprove()  — стандартный flow
// 2. test_SwapExactWithPermit()   — gasless flow
// Проверьте, что баланс DEX увеличился, а баланс user уменьшился на amountIn

Критерий приёмки: оба теста проходят; тест swapExactWithPermit не требует предварительного approve от пользователя.
