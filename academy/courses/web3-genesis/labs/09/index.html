Lab 09 · wagmi: React dApp

wagmi — это коллекция React-хуков для работы с EVM, построенная поверх viem. Он берёт на себя управление состоянием подключения кошелька, кэширование запросов через TanStack Query и реактивные обновления при смене аккаунта или сети. RainbowKit добавляет готовый UI-компонент «Connect Wallet» с поддержкой 50+ кошельков. В этом лабе вы соберёте полноценный React dApp: подключение MetaMask, чтение баланса, отправка трансфера, real-time события и управление сетью.

Цели лаба. После выполнения всех заданий вы сможете: настраивать WagmiProvider с RainbowKit; подключать MetaMask и читать данные аккаунта; использовать useReadContract для отображения баланса ERC-20; отправлять транзакции через useWriteContract и отслеживать их статус; подписываться на real-time события контракта; строить UX с состояниями idle/pending/success/error; обрабатывать смену сети через useSwitchChain.

);

Задание 3

Используйте useReadContract для отображения баланса ERC-20 подключённого аккаунта. Добавьте автообновление.

src/components/TokenBalance.tsx

import { useReadContract, useReadContracts } from "wagmi";
import { formatUnits } from "viem";

const ERC20_ABI = [
  { name: "name",      type: "function", stateMutability: "view", inputs: [], outputs: [{ type: "string" }] },
  { name: "symbol",    type: "function", stateMutability: "view", inputs: [], outputs: [{ type: "string" }] },
  { name: "decimals",  type: "function", stateMutability: "view", inputs: [], outputs: [{ type: "uint8" }] },
  { name: "balanceOf", type: "function", stateMutability: "view",
    inputs: [{ name: "account", type: "address" }],
    outputs: [{ type: "uint256" }] },
] as const;

interface Props {
  tokenAddress: `0x${string}`;
  account: `0x${string}`;
}

export function TokenBalance({ tokenAddress, account }: Props) {
  // useReadContracts — batched multicall: один HTTP-запрос вместо четырёх
  const { data, isLoading, isError } = useReadContracts({
    contracts: [
      { address: tokenAddress, abi: ERC20_ABI, functionName: "name" },
      { address: tokenAddress, abi: ERC20_ABI, functionName: "symbol" },
      { address: tokenAddress, abi: ERC20_ABI, functionName: "decimals" },
      { address: tokenAddress, abi: ERC20_ABI, functionName: "balanceOf", args: [account] },
    ],
    // Автообновление каждые 12 секунд (≈1 блок Ethereum):
    query: { refetchInterval: 12_000 },
  });

  if (isLoading) return <div className="card">Загрузка...</div>;
  if (isError || !data)  return <div className="card">Ошибка чтения контракта</div>;

  const [name, symbol, decimals, balance] = data.map((d) => d.result);

  return (
    <div className="card">
      <h3>{name as string} ({symbol as string})</h3>
      <p>Ваш баланс: {formatUnits(balance as bigint, decimals as number)} {symbol as string}</p>
      <small>Адрес: {account}</small>
    </div>
  );
}

Задание 4

Реализуйте форму трансфера с useWriteContract и useWaitForTransactionReceipt.

src/components/TransferForm.tsx

import { useState } from "react";
import { useWriteContract, useWaitForTransactionReceipt } from "wagmi";
import { parseUnits, isAddress } from "viem";

const ERC20_ABI = [
  {
    name: "transfer",
    type: "function",
    stateMutability: "nonpayable",
    inputs: [
      { name: "to",    type: "address" },
      { name: "value", type: "uint256" },
    ],
    outputs: [{ type: "bool" }],
  },
] as const;

interface Props { tokenAddress: `0x${string}`; }

export function TransferForm({ tokenAddress }: Props) {
  const [to, setTo]         = useState("");
  const [amount, setAmount] = useState("");

  // Шаг 1: writeContract отправляет транзакцию
  const { writeContract, data: txHash, isPending, error } = useWriteContract();

  // Шаг 2: waitForTransactionReceipt ожидает включения в блок
  const { isLoading: isConfirming, isSuccess } = useWaitForTransactionReceipt({
    hash: txHash,
  });

  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    if (!isAddress(to)) { alert("Неверный адрес"); return; }

    writeContract({
      address: tokenAddress,
      abi: ERC20_ABI,
      functionName: "transfer",
      args: [to as `0x${string}`, parseUnits(amount, 6)],
    });
  };

  return (
    <div className="card">
      <h3>Отправить токены</h3>
      <form onSubmit={handleSubmit}>
        <input
          placeholder="Адрес получателя (0x...)"
          value={to}
          onChange={(e) => setTo(e.target.value)}
          required
        />
        <input
          type="number"
          placeholder="Сумма (USDC)"
          value={amount}
          onChange={(e) => setAmount(e.target.value)}
          min="0.000001"
          step="0.000001"
          required
        />
        <button type="submit" disabled={isPending || isConfirming}>
          {isPending     ? "Подтвердите в кошельке..." :
           isConfirming  ? "Ожидание блока..." :
           "Отправить"}
        </button>
      </form>

      {txHash   && <p>Tx: <a href={`https://sepolia.etherscan.io/tx/${txHash}`} target="_blank">{txHash.slice(0,10)}...</a></p>}
      {isSuccess && <p className="success">Трансфер подтверждён!</p>}
      {error    && <p className="error">{error.message}</p>}
    </div>
  );
}

Задание 5

Подпишитесь на события Transfer в реальном времени с помощью useWatchContractEvent.

src/components/EventFeed.tsx

import { useState } from "react";
import { useWatchContractEvent } from "wagmi";
import { formatUnits } from "viem";

const TRANSFER_ABI = [
  {
    name: "Transfer",
    type: "event",
    inputs: [
      { name: "from",  type: "address", indexed: true },
      { name: "to",    type: "address", indexed: true },
      { name: "value", type: "uint256", indexed: false },
    ],
  },
] as const;

interface TransferEvent {
  from: string;
  to:   string;
  value: bigint;
  blockNumber: bigint;
}

interface Props { tokenAddress: `0x${string}`; }

export function EventFeed({ tokenAddress }: Props) {
  const [events, setEvents] = useState<TransferEvent[]>([]);

  // useWatchContractEvent создаёт WebSocket или polling подписку
  useWatchContractEvent({
    address: tokenAddress,
    abi: TRANSFER_ABI,
    eventName: "Transfer",
    onLogs(logs) {
      const newEvents = logs.map((log) => ({
        from:        log.args.from!,
        to:          log.args.to!,
        value:       log.args.value!,
        blockNumber: log.blockNumber!,
      }));
      // Добавляем новые события в начало списка (максимум 20):
      setEvents((prev) => [...newEvents, ...prev].slice(0, 20));
    },
  });

  return (
    <div className="card">
      <h3>Последние трансферы (live)</h3>
      {events.length === 0 && <p>Ожидание событий...</p>}
      <ul>
        {events.map((ev, i) => (
          <li key={i}>
            Блок {ev.blockNumber.toString()}:
            {ev.from.slice(0,8)}... → {ev.to.slice(0,8)}...
            <strong> {formatUnits(ev.value, 6)} USDC</strong>
          </li>
        ))}
      </ul>
    </div>
  );
}

Задание 7

Обработайте ситуацию «неправильная сеть»: если пользователь подключён не к Sepolia — показать кнопку переключения через useSwitchChain.

src/components/NetworkGuard.tsx

import { useAccount, useChainId, useSwitchChain } from "wagmi";
import { sepolia } from "wagmi/chains";
import type { ReactNode } from "react";

interface Props { children: ReactNode; }

export function NetworkGuard({ children }: Props) {
  const chainId = useChainId();
  const { isConnected } = useAccount();
  const { switchChain, isPending } = useSwitchChain();

  // Если кошелёк не подключён — показываем children (App сам обработает)
  if (!isConnected) return <>{children}</>;

  // Если подключены к нужной сети — всё хорошо
  if (chainId === sepolia.id) return <>{children}</>;

  // Иначе — предупреждение и кнопка переключения
  return (
    <div className="network-warning">
      <p>
        Вы подключены к сети с chainId {chainId}.
        Это приложение работает только с <strong>Sepolia (chainId 11155111)</strong>.
      </p>
      <button
        onClick={() => switchChain({ chainId: sepolia.id })}
        disabled={isPending}
      >
        {isPending ? "Переключение..." : "Переключиться на Sepolia"}
      </button>
    </div>
  );
}

useSwitchChain вызывает wallet_switchEthereumChain (EIP-3326). Если сеть не добавлена в MetaMask — автоматически предложит wallet_addEthereumChain. Конфигурация цепочки (RPC URL, название, нативная валюта) берётся из объекта sepolia пакета viem/chains.

wagmi v2 требует viem v2 и TanStack Query v5 как peer-зависимости. RainbowKit v2 поддерживает только wagmi v2. При установке проверяйте совместимость версий. Полная документация: wagmi.sh и rainbowkit.com.
