Security: Reentrancy, Checks-Effects-Interactions и инварианты

→ продолжение Lesson 08 — The Graph. Если индексер показывает «странный» rebalance — это может быть симптом атаки. Здесь учимся писать контракты, которые в принципе не дают злоумышленнику войти повторно во внешний вызов.

Re-entrancy — это самый дорогой класс багов в истории Ethereum. В 2016 году The DAO потеряла $60M (по курсу того дня), что привело к hard fork и появлению Ethereum Classic. Параллельная индустрия аудита, инструменты статического анализа (Slither, Mythril), формальной верификации (Certora) и bug-bounty (Immunefi) — всё это во многом результат уроков этой одной уязвимости. И тем не менее, новые случаи всплывают каждый месяц: Cream Finance ($130M, 2021), Fei Protocol ($80M, 2022), Curve ($73M, 2023).

Что вы получите. После этого урока вы умеете: распознать reentrancy-вектор по форме функции; написать функцию по паттерну Checks-Effects-Interactions; применить ReentrancyGuard; отличить single-function, cross-function и read-only reentrancy; провести базовую проверку контрактного кода через Slither.

1. Природа re-entrancy

Каждый раз, когда контракт делает внешний вызов на адрес, который может содержать код, контроль передаётся туда. Если злоумышленник управляет этим адресом, его контракт может снова вызвать оригинальный контракт — до того, как тот успел обновить своё состояние. Это и есть reentrancy. 

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

// УЯЗВИМЫЙ кошелёк — повторяет архитектуру The DAO
contract VulnerableVault {
    mapping(address => uint256) public balances;

    function deposit() external payable {
        balances[msg.sender] += msg.value;
    }

    function withdraw() external {
        uint256 bal = balances[msg.sender];
        require(bal > 0, "no balance");

        // ШАГ 1: внешний вызов — отдаём ETH
        (bool ok, ) = msg.sender.call{value: bal}("");
        require(ok, "transfer failed");

        // ШАГ 2: обновляем state ПОСЛЕ внешнего вызова — фатально
        balances[msg.sender] = 0;
    }
}

Атакующий деплоит контракт, у которого в receive() внутри снова дёргает withdraw() жертвы. На момент повторного вызова balances[attacker] ещё не обнулён — поэтому проверка require(bal > 0) проходит, и атакующий снова получает свою долю. Цикл повторяется, пока в Vault не закончится ETH или не закончится газ. 

// Контракт-атакующий
contract Attacker {
    VulnerableVault public vault;
    uint256 public iterations;

    constructor(address payable _vault) {
        vault = VulnerableVault(_vault);
    }

    function attack() external payable {
        require(msg.value >= 1 ether, "send seed");
        vault.deposit{value: 1 ether}();
        vault.withdraw();
    }

    // Сюда приходит ETH из vault.withdraw() — здесь мы атакуем снова
    receive() external payable {
        if (address(vault).balance >= 1 ether && iterations < 10) {
            iterations++;
            vault.withdraw(); // повторный вход!
        }
    }
}

2. Паттерн Checks-Effects-Interactions (CEI)

Простейшее и самое эффективное лекарство — изменить порядок операций. Сначала проверки, потом обновление state, потом — внешний вызов. 

function withdraw() external {
    // 1. CHECKS — все require/проверки сверху
    uint256 bal = balances[msg.sender];
    require(bal > 0, "no balance");

    // 2. EFFECTS — обновляем state ДО любого внешнего вызова
    balances[msg.sender] = 0;

    // 3. INTERACTIONS — теперь можно делать внешние вызовы
    (bool ok, ) = msg.sender.call{value: bal}("");
    require(ok, "transfer failed");
}

Когда атакующий вернётся в withdraw() через receive(), его balances[attacker] уже равен нулю, и require(bal > 0) завершит транзакцию с revert. Все внесённые им изменения отменятся (включая повторный вход).

Почему именно .call, а не .transfer? Раньше Solidity-сообщество рекомендовало address.transfer(value) и address.send(value) как «безопасные» из-за лимита 2300 газа. Но после EIP-1884 (Istanbul) стоимость SLOAD выросла, и многие нормальные контракты-кошельки перестали умещаться в 2300 газа. Теперь стандарт — .call{value: x}("") + CEI/ReentrancyGuard.

3. ReentrancyGuard — страховка

OpenZeppelin предоставляет модификатор nonReentrant, который ставит флаг при входе в функцию и снимает его при выходе. Повторный вход проваливается на этом флаге. 

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

import "@openzeppelin/contracts/utils/ReentrancyGuard.sol";

contract SafeVault is ReentrancyGuard {
    mapping(address => uint256) public balances;

    function deposit() external payable {
        balances[msg.sender] += msg.value;
    }

    function withdraw() external nonReentrant {
        uint256 bal = balances[msg.sender];
        require(bal > 0, "no balance");
        balances[msg.sender] = 0;
        (bool ok, ) = msg.sender.call{value: bal}("");
        require(ok, "transfer failed");
    }
}

Под капотом ReentrancyGuard устанавливает приватную переменную _status в значение _ENTERED при входе и _NOT_ENTERED при выходе. Повторный вход не пройдёт проверку. Цена одного использования модификатора — около 5 000 газа (один SSTORE на изменение слота с ненулевого на ненулевое и обратно).

OpenZeppelin 5.x: ReentrancyGuardTransient. В Cancun EIP-1153 добавил TSTORE/TLOAD — transient storage, который сбрасывается в конце транзакции. ReentrancyGuardTransient использует именно его и стоит ~200 газа вместо 5 000. Доступен с Solidity 0.8.24 и OpenZeppelin 5.1.

4. Cross-function reentrancy

Реентранс может быть не только в ту же функцию, но и через другую. Если две функции читают/пишут один и тот же state, и одна из них защищена nonReentrant, но другая — нет, атакующий войдёт через вторую. 

contract CrossFunctionVulnerable is ReentrancyGuard {
    mapping(address => uint256) public balances;

    function withdrawAll() external nonReentrant {
        uint256 bal = balances[msg.sender];
        (bool ok, ) = msg.sender.call{value: bal}("");
        require(ok);
        balances[msg.sender] = 0;
    }

    // Эта функция НЕ защищена — но читает тот же balances:
    function transferInternal(address to, uint256 amount) external {
        require(balances[msg.sender] >= amount, "low");
        balances[msg.sender] -= amount;
        balances[to] += amount;
    }
}

// Атака: внутри receive() — вызвать transferInternal, перевести
// «несуществующие» (уже отданные нам) деньги другу. ReentrancyGuard
// не помогает, потому что transferInternal не помечен nonReentrant.

Правило: все функции, читающие общий критический state, должны быть либо помечены nonReentrant, либо следовать CEI. В нашем примере исправляется одним из двух способов: либо обновить state ДО внешнего вызова в withdrawAll, либо повесить nonReentrant на transferInternal.

5. Read-only reentrancy

Самый коварный вид. Атакующий не пишет в state жертвы — он читает. Многие протоколы (Curve, Balancer) используют функции типа get_virtual_price() как источник правды. Если во время внешнего вызова pool находится в «промежуточном» состоянии (часть инвариантов уже нарушена), view-функция вернёт некорректное значение. А контракт-потребитель (например, кредитная платформа) примет решение на основе этой «лживой» цены. 

// Уязвимый pool (упрощённо)
contract Pool {
    uint256 public reserve0;
    uint256 public reserve1;
    uint256 public totalLP;

    function getPrice() external view returns (uint256) {
        return reserve0 * 1e18 / totalLP;
    }

    function removeLiquidity(uint256 lp) external {
        uint256 amount0 = lp * reserve0 / totalLP;
        uint256 amount1 = lp * reserve1 / totalLP;

        // ВНЕШНИЙ вызов — отправка токена0 пользователю
        IERC20(token0).transfer(msg.sender, amount0); // ← reentrancy hook!

        // На этот момент:
        //   reserve0 ещё содержит старое значение (большое),
        //   но реально на балансе уже меньше.
        // totalLP всё ещё old.
        // Если token0 — ERC-777 (с callback), атакующий
        // вызовет getPrice() — получит inflated число.

        reserve0 -= amount0;
        reserve1 -= amount1;
        totalLP  -= lp;
    }
}
Реальный кейс. Cream Finance (2021): использовал Curve как oracle для крипто-ETH (cyETH). Curve get_virtual_price во время remove_liquidity возвращал inflated цену, потому что callback в ERC-777-токене вызывался ДО обновления баланса pool. Атакующий взял flash loan, depositнул, дернул колбэк, занял огромную сумму под «inflated» залог. Убыток — $130M. Лечение: nonReentrant на read-only функциях, либо state-locked модификатор, возвращающий revert если pool «in flight».

6. Где ещё прячется внешний вызов

Опасны все вызовы, которые передают контроль на адрес, не контролируемый разработчиком. Это:

  • address.call / transfer / send — ETH-перевод. Получатель может быть контрактом с receive()/fallback().
  • IERC20(token).transfer(...) — если токен это ERC-777 или вообще «hooks-aware», получатель получит tokensReceived. WETH, USDC — обычные ERC-20 без hook'ов, но любой неизвестный токен — потенциальная мина.
  • IERC721(nft).safeTransferFrom — обязан вызывать onERC721Received у получателя. Это вектор для NFT-протоколов (атака на Akutars, $34M, 2022).
  • delegatecall на не-trusted адрес — даёт абсолютный контроль над storage.
  • IERC1155(...).safeTransferFrom и любые callbacks стандарта.

7. Другие checks, без которых код небезопасен

7.1 Integer overflow

Solidity ≥0.8 включает overflow-checks по умолчанию (revert при выходе за пределы типа). До 0.8 — SafeMath. Опасный антипаттерн: явный unchecked { ... } ради экономии газа. 

function withdraw(uint256 amount) external {
    unchecked {
        balances[msg.sender] -= amount; // нет revert при underflow!
    }
}

// Правило: unchecked только если математически доказали отсутствие
// overflow. Например, после require(amount <= balances[msg.sender]).

7.2 tx.origin vs msg.sender

tx.origin — это исходный EOA, инициировавший транзакцию. msg.sender — непосредственный вызывающий. Если контракт A использует tx.origin для авторизации, атакующий может выманить жертву в свой контракт B, который вызовет A. tx.origin будет жертвой, авторизация пройдёт. 

// ОПАСНО
function transferOwnership(address newOwner) external {
    require(tx.origin == owner, "not owner"); // BAD
    owner = newOwner;
}

// БЕЗОПАСНО
function transferOwnership(address newOwner) external {
    require(msg.sender == owner, "not owner"); // GOOD
    owner = newOwner;
}

7.3 Front-running и MEV

Mempool публичен. Атакующий видит вашу транзакцию до её включения и может выставить свою с большим gas price (или через flashbots-bundle). Защита: commit-reveal, slippage-параметры, private mempool (Flashbots Protect, MEV Blocker), на крайний случай — EIP-712 подписи с дедлайном.

7.4 Oracle manipulation

Источник цены — критический элемент. Никогда не используйте spot price AMM (getReserves) как oracle для крупных решений: атакующий двигает цену в одном блоке через flash loan и эксплуатирует ваш контракт. Используйте TWAP (time-weighted) или внешние oracle (Chainlink).

7.5 Unchecked external call return

// BAD: не проверяем return — call мог вернуть false без revert
msg.sender.call{value: amount}("");

// GOOD
(bool ok, ) = msg.sender.call{value: amount}("");
require(ok, "call failed");

// BAD для ERC-20: USDT не возвращает bool, многие токены ломают стандарт
IERC20(token).transfer(to, amount);

// GOOD: SafeERC20 из OpenZeppelin
import "@openzeppelin/contracts/token/ERC20/utils/SafeERC20.sol";
using SafeERC20 for IERC20;
IERC20(token).safeTransfer(to, amount);

8. Slither — статический анализ

Slither от Trail of Bits — open-source инструмент, который ловит ~80% типичных багов за секунды. Он понимает Solidity AST, строит control-flow граф и применяет 90+ детекторов. 

# Установка (требует Python 3.10+)
pip install slither-analyzer

# Запуск на Foundry-проекте
cd my-foundry-project
slither .

# Только конкретные детекторы
slither . --detect reentrancy-eth,reentrancy-no-eth,arbitrary-send-eth

# Игнорировать определённые findings (после ручной проверки)
echo "// slither-disable-next-line reentrancy-no-eth" > ...

# Вывод примера:
# VulnerableVault.withdraw() (src/VulnerableVault.sol#10-15)
#         External calls:
#         - (ok) = msg.sender.call{value: bal}() (src/...#12)
#         State variables written after the call(s):
#         - balances[msg.sender] = 0 (src/...#14)
#         Reference: https://github.com/crytic/slither/wiki/Detector-Documentation#reentrancy-vulnerabilities

9. Практические задачи

Задача 1 — Воспроизвести DAO-style атаку локально

Создайте Foundry-проект с VulnerableVault и Attacker. Запустите forge test, который депозитит 10 ETH от честного юзера, потом атакующий за один вызов выкачивает всё. 

forge init reentrancy-lab && cd reentrancy-lab
# Поместите код VulnerableVault + Attacker в src/

// test/Attack.t.sol
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

import "forge-std/Test.sol";
import "../src/VulnerableVault.sol";
import "../src/Attacker.sol";

contract AttackTest is Test {
    VulnerableVault vault;
    Attacker attacker;
    address honest = address(0x1234);

    function setUp() public {
        vault = new VulnerableVault();
        attacker = new Attacker(payable(address(vault)));

        vm.deal(honest, 10 ether);
        vm.deal(address(this), 1 ether);
    }

    function test_drain() public {
        // Честный пользователь депозитит 10 ETH
        vm.prank(honest);
        vault.deposit{value: 10 ether}();
        assertEq(address(vault).balance, 10 ether);

        // Атакующий сеет 1 ETH и забирает всё
        attacker.attack{value: 1 ether}();

        assertEq(address(vault).balance, 0, "vault drained");
        assertGt(address(attacker).balance, 10 ether, "attacker profited");
        console.log("Attacker stole:", address(attacker).balance);
    }
}

forge test --match-test test_drain -vvv

Задача 2 — Исправить vault и убедиться, что атака больше не проходит

Сделайте копию контракта SafeVault (CEI + nonReentrant). Перепишите тест: он должен ожидать revert при попытке атаки и проверить инвариант баланса. 

// test/AttackSafe.t.sol
contract AttackSafeTest is Test {
    SafeVault vault;
    Attacker attacker;
    address honest = address(0x1234);

    function setUp() public {
        vault = new SafeVault();
        attacker = new Attacker(payable(address(vault)));
        vm.deal(honest, 10 ether);
    }

    function test_attackReverts() public {
        vm.prank(honest);
        vault.deposit{value: 10 ether}();

        // Атака должна провалиться: либо на ReentrancyGuard,
        // либо на require(bal > 0) после обнуления баланса
        vm.expectRevert();
        attacker.attack{value: 1 ether}();

        // Vault не пострадал
        assertEq(address(vault).balance, 10 ether, "vault intact");
    }
}

forge test --match-test test_attackReverts -vvv

Задача 4 — Найти и исправить cross-function reentrancy

Ниже контракт StakingPool с тонкой ошибкой. Один из методов защищён nonReentrant, второй — нет. Найдите вектор атаки, напишите PoC-тест, потом исправьте. 

// src/StakingPool.sol
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

import "@openzeppelin/contracts/utils/ReentrancyGuard.sol";

contract StakingPool is ReentrancyGuard {
    mapping(address => uint256) public stakes;
    mapping(address => uint256) public rewards;

    function stake() external payable {
        stakes[msg.sender] += msg.value;
        rewards[msg.sender] += msg.value / 100; // 1% reward
    }

    // Защищена, но reward отдаётся ПЕРЕД обнулением
    function claimReward() external nonReentrant {
        uint256 r = rewards[msg.sender];
        require(r > 0, "no reward");
        (bool ok, ) = msg.sender.call{value: r}("");
        require(ok);
        rewards[msg.sender] = 0;
    }

    // НЕ защищена nonReentrant
    function transferStake(address to, uint256 amount) external {
        require(stakes[msg.sender] >= amount, "low");
        stakes[msg.sender] -= amount;
        stakes[to] += amount;
        // Также «переводим» долю reward
        rewards[to] += amount / 100;
    }
}

// ВЕКТОР АТАКИ:
// 1. Атакующий stake() 1 ETH → reward = 0.01 ETH
// 2. Атакующий dev claimReward()
// 3. В receive() ETH вызывает transferStake(friend, 1 ether)
//    — это разрешено (nonReentrant НЕ на transferStake), и
//    rewards[friend] += 0.01 ETH (новый незаработанный reward)
// 4. После claimReward() атакующий и friend имеют по 0.01 ETH
//    rewards — но в pool было только 0.01 ETH total

// ИСПРАВЛЕНИЕ:
//   - либо CEI: обнулить rewards[msg.sender] ДО внешнего вызова
//   - либо добавить nonReentrant на transferStake тоже

Запустите тест-PoC и затем исправление. Используйте vm.expectRevert() в финальном варианте для подтверждения, что патч закрывает уязвимость.

10. Песочница · уязвимый и безопасный vault в Remix

Iframe ниже открывает оба контракта (vulnerable + safe + attacker) в одном файле. В Remix → JavaScript VM задеплойте всё, переведите 1 ETH в VulnerableVault от Account[0], затем переключитесь на Account[1], деплойте Attacker и вызовите attack{1 ETH}. Посмотрите, как баланс vault уйдёт в attacker. Потом повторите с SafeVault — увидите revert. 

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

import "https://github.com/OpenZeppelin/openzeppelin-contracts/blob/v5.0.2/contracts/utils/ReentrancyGuard.sol";

contract VulnerableVault {
    mapping(address => uint256) public balances;
    function deposit() external payable { balances[msg.sender] += msg.value; }
    function withdraw() external {
        uint256 bal = balances[msg.sender];
        require(bal > 0, "no balance");
        (bool ok, ) = msg.sender.call{value: bal}("");
        require(ok);
        balances[msg.sender] = 0;
    }
    receive() external payable {}
}

contract SafeVault is ReentrancyGuard {
    mapping(address => uint256) public balances;
    function deposit() external payable { balances[msg.sender] += msg.value; }
    function withdraw() external nonReentrant {
        uint256 bal = balances[msg.sender];
        require(bal > 0, "no balance");
        balances[msg.sender] = 0;
        (bool ok, ) = msg.sender.call{value: bal}("");
        require(ok);
    }
    receive() external payable {}
}

contract Attacker {
    VulnerableVault public vault;
    uint256 public iter;
    constructor(address payable _v) { vault = VulnerableVault(_v); }
    function attack() external payable {
        vault.deposit{value: msg.value}();
        vault.withdraw();
    }
    receive() external payable {
        if (address(vault).balance >= 1 ether && iter < 10) {
            iter++;
            vault.withdraw();
        }
    }
    function take() external {
        payable(msg.sender).transfer(address(this).balance);
    }
}
