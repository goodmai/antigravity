Chainlink Oracles — внешние данные для смарт-контрактов

→ продолжение Lesson 08 — The Graph. Там мы научились читать on-chain данные эффективно. Здесь — учимся приводить в on-chain данные, которых на блокчейне нет: цены, случайные числа, веб-API, кросс-чейн сообщения.

EVM детерминирован: одна транзакция — одно состояние, одинаковое у всех нод. Это значит, что контракт не может просто так сделать HTTP-запрос: разные ноды получат разные ответы и сеть расколется. Чтобы безопасно «прибить» внешний мир к консенсусу, нужен оракул — отдельная сеть, которая собирает данные, криптографически подписывает агрегированный результат и публикует его on-chain. Chainlink — крупнейшая такая сеть: на 2025 она секурит ~$22 трлн совокупного объёма (TVS).

Что вы получите. После урока вы умеете: подключить Chainlink Data Feed в Solidity и читать его на фронте; получить verifiable-random number через VRF v2.5 для NFT-mint'а или лотереи; запланировать периодические задачи через Chainlink Automation (бывший Keepers); сделать кросс-чейн вызов через CCIP; вызвать произвольный HTTP API из контракта через Chainlink Functions; защищать контракты от stale price и L2-sequencer downtime.

1. Почему контракт не может сделать HTTP-запрос

Каждый full-node Ethereum перепроверяет все транзакции в блоке. Если бы EVM-инструкция HTTP существовала, то нода A в Германии получила бы ответ «1.23», а нода B в Сингапуре — «1.24» (разный latency, разная кеш-вершина, разный round-robin DNS). Хэши блоков разошлись бы — сеть расколота.

Решение универсальное: внешние данные приходят как обычная on-chain транзакция, которую отправляет оракул. EVM просто читает их из storage чужого контракта — это детерминировано.

│                       │                                          │
└───────────────────────┼──────────────────────────────────────────┘
                        │  ONE tx on-chain

              ┌────────────────────────┐
              │  Aggregator contract   │   ← вы читаете отсюда
              │  ETH/USD .latestRound  │
              └────────────────────────┘

Что важно знать на проде:

  • Heartbeat — максимальное время между апдейтами. Для ETH/USD на mainnet — 3600 с. Если данные тише, чем deviation threshold (~0.5%), feed всё равно обновится раз в час.
  • Deviation threshold — если цена «убежала» на N% от последнего опубликованного, апдейт случится сразу.
  • decimals — у разных feed'ов разное. ETH/USD = 8, AAVE/ETH = 18. Никогда не хардкодьте.
  • Stale-check — обязательно сравнивать updatedAt с block.timestamp. На бридже / ликвидации старая цена = подарок арбитражёру.

4. Чтение Data Feed — Solidity

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

import { AggregatorV3Interface }
    from "@chainlink/contracts/src/v0.8/shared/interfaces/AggregatorV3Interface.sol";

contract EthUsdReader {
    AggregatorV3Interface public immutable feed;
    uint256 public constant STALE_AFTER = 1 hours;     // должно быть >= heartbeat

    error StalePrice(uint256 updatedAt, uint256 now_);
    error InvalidPrice(int256 answer);

    constructor(address _feed) {
        // Mainnet ETH/USD = 0x5f4eC3Df9cbd43714FE2740f5E3616155c5b8419
        feed = AggregatorV3Interface(_feed);
    }

    /// @return price нормализованная цена в 18 decimals
    function ethUsd() external view returns (uint256 price) {
        (
            uint80 roundId,
            int256 answer,
            ,
            uint256 updatedAt,
            uint80 answeredInRound
        ) = feed.latestRoundData();

        if (answer <= 0)                      revert InvalidPrice(answer);
        if (updatedAt == 0 || block.timestamp - updatedAt > STALE_AFTER)
            revert StalePrice(updatedAt, block.timestamp);
        if (answeredInRound < roundId)        revert StalePrice(updatedAt, block.timestamp);

        uint8 d = feed.decimals();             // 8 для ETH/USD
        price = uint256(answer) * (10 ** (18 - d));
    }
}

Это минимальный production-grade чтения. Все три проверки (answer > 0, updatedAt, answeredInRound) — не «paranoid», а буквально условия безопасной интеграции, описанные в Chainlink API Reference.

5. Чтение Data Feed — viem (JS-фронт)

import { createPublicClient, http, parseAbi } from "viem";
import { mainnet } from "viem/chains";

const ETH_USD = "0x5f4eC3Df9cbd43714FE2740f5E3616155c5b8419";
const ABI = parseAbi([
  "function latestRoundData() view returns (uint80, int256, uint256, uint256, uint80)",
  "function decimals() view returns (uint8)",
  "function description() view returns (string)",
]);

const client = createPublicClient({ chain: mainnet, transport: http() });

const [decimals, desc, round] = await Promise.all([
  client.readContract({ address: ETH_USD, abi: ABI, functionName: "decimals" }),
  client.readContract({ address: ETH_USD, abi: ABI, functionName: "description" }),
  client.readContract({ address: ETH_USD, abi: ABI, functionName: "latestRoundData" }),
]);

const [, answer, , updatedAt] = round;
const price = Number(answer) / 10 ** decimals;
const age   = Math.floor(Date.now() / 1000) - Number(updatedAt);

console.log(`${desc} = $${price.toFixed(2)}  (age: ${age}s)`);
// → ETH / USD = $2411.50  (age: 17s)

6. L2-специфика: Sequencer Uptime Feed

На Arbitrum / Optimism / Base sequencer — единственная сторона, которая включает транзакции. Если sequencer падает, цены Chainlink на L2 замораживаются на старом значении (потому что никто не может отправить апдейт-tx). Это смертельно для ликвидаций: пока sequencer лежит, пользователи не могут добавить коллатерал, а цена «не движется» — после восстановления половина протокола ликвидируется.

Chainlink даёт специальный Sequencer Uptime Feed, который надо обязательно проверять перед использованием цен на L2: 

// На Arbitrum: 0xFdB631F5EE196F0ed6FAa767959853A9F217697D
AggregatorV3Interface internal sequencerUptimeFeed;
uint256 private constant GRACE_PERIOD = 3600;     // 1 час после восстановления

modifier onlyWhenSequencerUp() {
    (, int256 answer, uint256 startedAt, , ) = sequencerUptimeFeed.latestRoundData();
    if (answer == 1)       revert SequencerDown();           // 0 = up, 1 = down
    if (block.timestamp - startedAt <= GRACE_PERIOD)
        revert SequencerJustResumed();
    _;
}

function liquidate(address user) external onlyWhenSequencerUp { /* ... */ }

Grace period — это окно, в течение которого после восстановления sequencer'а ликвидации запрещены: пользователям нужно время «вернуться» и скорректировать позиции.

7. VRF v2.5: проверяемая случайность

block.timestamp, blockhash, keccak256 от prevhash + сендера — не случайны. Майнер / валидатор / sequencer может перебирать порядок транзакций или вообще не включать «невыгодные» блоки, выбирая удобный исход. Если на кону есть деньги (NFT-mint, лотерея, mystery box), это эксплойт.

Chainlink VRF комбинирует приватный ключ оракула и seed запроса в BLS-подпись; результат хешируется в случайное число и приходит вместе с криптографическим доказательством, которое проверяется on-chain. Манипулировать им может только тот, кто контролирует приватный ключ оракула — а это не вы и не валидатор. 

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

import { VRFConsumerBaseV2Plus } from
    "@chainlink/contracts/src/v0.8/vrf/dev/VRFConsumerBaseV2Plus.sol";
import { VRFV2PlusClient } from
    "@chainlink/contracts/src/v0.8/vrf/dev/libraries/VRFV2PlusClient.sol";

contract FairNFT is VRFConsumerBaseV2Plus {
    uint256 public immutable subId;          // Chainlink subscription
    bytes32 public immutable keyHash;        // gas-lane
    uint16  public constant CONFIRMATIONS = 3;
    uint32  public constant CALLBACK_GAS  = 200_000;
    uint32  public constant NUM_WORDS     = 1;

    mapping(uint256 => address) public requestToMinter;
    mapping(address => uint256) public minterToTokenId;

    event MintRequested(uint256 requestId, address minter);
    event MintRevealed(uint256 requestId, address minter, uint256 tokenId);

    constructor(address vrfCoordinator, uint256 _subId, bytes32 _keyHash)
        VRFConsumerBaseV2Plus(vrfCoordinator)
    {
        subId = _subId;
        keyHash = _keyHash;
    }

    function requestMint() external returns (uint256 requestId) {
        requestId = s_vrfCoordinator.requestRandomWords(
            VRFV2PlusClient.RandomWordsRequest({
                keyHash:              keyHash,
                subId:                subId,
                requestConfirmations: CONFIRMATIONS,
                callbackGasLimit:     CALLBACK_GAS,
                numWords:             NUM_WORDS,
                extraArgs:            VRFV2PlusClient._argsToBytes(
                    VRFV2PlusClient.ExtraArgsV1({ nativePayment: false })
                )
            })
        );
        requestToMinter[requestId] = msg.sender;
        emit MintRequested(requestId, msg.sender);
    }

    function fulfillRandomWords(uint256 requestId, uint256[] calldata words)
        internal override
    {
        address minter = requestToMinter[requestId];
        uint256 tokenId = words[0] % 10_000;        // 10k коллекция
        minterToTokenId[minter] = tokenId;
        emit MintRevealed(requestId, minter, tokenId);
    }
}
Commit–reveal vs VRF. Раньше «честный» mint делали через commit–reveal в две транзакции (юзер commit'ит hash, потом reveal'ит pre-image). Проблема: юзер может «не reveal'ить», если ему не повезло. VRF решает обе стороны: и dApp не может крутить, и юзер не может отменить — результат предопределён в момент запроса.

8. Automation (бывший Keepers): cron on-chain

Сам по себе блокчейн не «вызывает» функции — кто-то должен отправить транзакцию. Если ваш протокол требует ежедневного ребаланса, accrue процентов, расфасовки vesting'ов, кто-то должен дёргать функцию каждые сутки. Сделать «крон» — это нанять keeper-сеть. Chainlink Automation — самая дешёвая и надёжная. 

import { AutomationCompatibleInterface }
    from "@chainlink/contracts/src/v0.8/automation/AutomationCompatible.sol";

contract DailyRebalancer is AutomationCompatibleInterface {
    uint256 public lastRun;
    uint256 public constant INTERVAL = 1 days;

    /// view-функция: должна ли сеть дёрнуть performUpkeep сейчас?
    function checkUpkeep(bytes calldata /* checkData */)
        external view override
        returns (bool upkeepNeeded, bytes memory performData)
    {
        upkeepNeeded = (block.timestamp - lastRun) >= INTERVAL;
        performData  = abi.encode(block.timestamp);
    }

    /// фактический ребаланс — будет вызвана keeper'ом, когда upkeepNeeded = true
    function performUpkeep(bytes calldata performData) external override {
        if (block.timestamp - lastRun < INTERVAL) revert("not yet");
        lastRun = block.timestamp;
        _rebalance();
    }

    function _rebalance() internal { /* swap, supply, withdraw */ }
}

Регистрация upkeep'а — в automation.chain.link: указываете адрес контракта, минимальный газ, гарантированный баланс LINK (≈5 LINK хватит на месяцы). С этого момента сеть Chainlink в каждом блоке симулирует ваш checkUpkeep off-chain (бесплатно) и вызывает performUpkeep on-chain, когда нужно.

Триггеры — три варианта:

  • Custom Logic — как выше: ваш checkUpkeep решает «пора ли».
  • Time-based — cron-выражение (0 12 * * *), без своего checkUpkeep вообще.
  • Log Trigger — реакция на событие конкретного контракта; performUpkeep получает данные лога.

9. CCIP: cross-chain messaging

Если на Ethereum mainnet хранится governance, а реальное приложение работает на Arbitrum — нужен надёжный канал передачи сообщений. Chainlink CCIP (Cross-Chain Interoperability Protocol) — это управляемый Chainlink-сетью «мост сообщений» с двумя свойствами:

  • Программный, как call на другую сеть. Можно передать и сообщение, и токены, и то и другое.
  • Защита «Risk Management Network» — отдельный набор нод (DON), который может заблокировать подозрительный мессадж до ручного разбора.
import { IRouterClient } from
    "@chainlink/contracts-ccip/src/v0.8/ccip/interfaces/IRouterClient.sol";
import { Client } from
    "@chainlink/contracts-ccip/src/v0.8/ccip/libraries/Client.sol";

contract Sender {
    IRouterClient public immutable router;
    address public immutable linkToken;

    constructor(address _router, address _link) {
        router = IRouterClient(_router);
        linkToken = _link;
    }

    function sendMessage(
        uint64 destChainSelector,           // напр. 4949039107694359620 = Arbitrum
        address receiver,
        string calldata text
    ) external returns (bytes32 messageId) {
        Client.EVM2AnyMessage memory msg_ = Client.EVM2AnyMessage({
            receiver: abi.encode(receiver),
            data:     abi.encode(text),
            tokenAmounts: new Client.EVMTokenAmount[](0),
            extraArgs: Client._argsToBytes(
                Client.GenericExtraArgsV2({
                    gasLimit: 200_000,
                    allowOutOfOrderExecution: true
                })
            ),
            feeToken: linkToken            // 0x0 = pay в native
        });

        uint256 fee = router.getFee(destChainSelector, msg_);
        IERC20(linkToken).approve(address(router), fee);
        messageId = router.ccipSend(destChainSelector, msg_);
    }
}
// На стороне получателя (другая сеть):
import { CCIPReceiver } from
    "@chainlink/contracts-ccip/src/v0.8/ccip/applications/CCIPReceiver.sol";

contract Receiver is CCIPReceiver {
    string public lastMessage;
    address public lastSender;
    uint64  public lastSourceChain;

    constructor(address router) CCIPReceiver(router) {}

    function _ccipReceive(Client.Any2EVMMessage memory m) internal override {
        lastSourceChain = m.sourceChainSelector;
        lastSender      = abi.decode(m.sender, (address));
        lastMessage     = abi.decode(m.data,   (string));
    }
}

10. Functions: HTTP-вызов из контракта

Chainlink Functions — это «serverless для смарт-контрактов». Вы пишете JS-код, который исполнится в децентрализованной сети нод (каждая в своём песочнице Deno), результат агрегируется и возвращается в callback контракта. 

// Solidity вызывающий
import { FunctionsClient } from
    "@chainlink/contracts/src/v0.8/functions/v1_0_0/FunctionsClient.sol";
import { FunctionsRequest } from
    "@chainlink/contracts/src/v0.8/functions/v1_0_0/libraries/FunctionsRequest.sol";

contract WeatherDeriv is FunctionsClient {
    using FunctionsRequest for FunctionsRequest.Request;

    bytes32 public lastRequestId;
    string  public lastResponse;
    uint64  public immutable subId;
    bytes32 public immutable donId;

    constructor(address router, uint64 _subId, bytes32 _donId)
        FunctionsClient(router)
    {
        subId = _subId; donId = _donId;
    }

    function requestTemperature(string calldata city) external returns (bytes32) {
        FunctionsRequest.Request memory r;
        r.initializeRequestForInlineJavaScript(
            "const city = args[0];"
            "const r = await Functions.makeHttpRequest({"
            "  url: `https://api.open-meteo.com/v1/forecast?city=${city}¤t=temperature_2m`,"
            "});"
            "if (r.error) throw Error('api fail');"
            "return Functions.encodeUint256(Math.round(r.data.current.temperature_2m * 100));"
        );
        string[] memory args = new string[](1);
        args[0] = city;
        r.setArgs(args);

        lastRequestId = _sendRequest(r.encodeCBOR(), subId, 300_000, donId);
        return lastRequestId;
    }

    function fulfillRequest(bytes32 requestId, bytes memory response, bytes memory /* err */)
        internal override
    {
        require(requestId == lastRequestId, "wrong req");
        lastResponse = string(response);
    }
}

Тонкость: каждая нода исполняет JS независимо. Если API отдаёт нестабильные данные (timestamp, random), результаты разойдутся и запрос упадёт. Поэтому в Functions-коде делают квантование: округляют, кропают точность, фиксируют timestamp на блочный.

13. Четыре практические задачи

13.1 Задача №1 — «Конвертер» с защитой от stale

Напишите контракт Converter, который принимает uint256 amountWei и возвращает его эквивалент в USD (18 decimals). Требования:

  • Использовать ETH / USD feed.
  • revert StalePrice, если updatedAt старше 1 часа на mainnet (или старше 24 часов на L2 с учётом sequencer-feed).
  • Тест: ввести моковый feed с updatedAt = 0 — должен упасть.
  • Тест: ввести feed с answer = -1 — должен упасть.
  • Тест на Anvil + fork mainnet: cast call <Converter> "convertEthToUsd(uint256)(uint256)" 1ether — вернуть ~$2400e18 (в зависимости от текущей цены).

13.2 Задача №2 — Честная лотерея на VRF

Контракт Lottery:

  • Пользователи покупают билеты за 0.01 ETH, ограничение maxTickets = 100.
  • После заполнения drawWinner() запрашивает VRF.
  • В fulfillRandomWords выбирается победитель winner = participants[words[0] % participants.length]; ему уходит 95% пула, 5% — owner'у на subscription LINK.
  • Контракт не должен позволять drawWinner дважды или buyTicket после drawWinner.
  • Юнит-тесты на форке Sepolia + реальная VRF v2.5 subscription (получите тестовый LINK на faucets.chain.link).

Бонусное условие: реализуйте refund, если за 7 дней fulfillRandomWords не пришёл (например, кончился LINK в subscription). См. VRF v2.5 billing.

13.3 Задача №3 — Auto-compounder через Automation

Контракт AaveAutoCompounder, который:

  • Хранит aUSDC-позицию пользователей (одна на адрес).
  • Каждые 24 часа Automation вызывает performUpkeep, который тащит accumulated rewards и реинвестирует в Aave Supply.
  • checkUpkeep возвращает true, только если rewards > minRewards = 10 USDC (иначе газ > profit).

13.4 Задача №4 — Cross-chain голосование через CCIP

Реализуйте две части:

  • На Sepolia: контракт Governor, в котором владельцы голосования (по сценарию simple-multisig) принимают решение «отправить N USDC на Arbitrum Sepolia в адрес Treasury». После голосования execute() вызывает ccipSend.
  • На Arbitrum Sepolia: контракт Treasury наследует CCIPReceiver, принимает (address to, uint256 amount) из _ccipReceive и отправляет USDC.
  • Тест: запустите цикл end-to-end. Проверьте трекинг message-id в ccip.chain.link explorer.
  • Безопасность: на стороне receiver'а проверяйте sourceChainSelector и sender — иначе любая нода может отправить «вам» произвольное сообщение.

14. Чек-лист «безопасная интеграция оракула»

  • Всегда проверяйте answer > 0, updatedAt > 0, block.timestamp − updatedAt ≤ heartbeat × 1.1, answeredInRound ≥ roundId.
  • На L2 — Sequencer Uptime Feed + grace period.
  • Никогда не используйте blockhash / block.timestamp для значений с экономическим эффектом — это VRF-кейс.
  • Адреса feed'ов — через constructor / config, не hardcode.
  • Хеджируйте на двух оракулах, если позиция большая (Chainlink + Pyth, либо два независимых feed'а): отдавайте min(feed1, feed2) для занижения LTV.
  • Для Automation: контракт должен корректно работать, если performUpkeep вызвали с опозданием, или вообще вручную — никаких «однократных» инвариантов.
  • Для CCIP / Functions: allowlist отправителей и источников. Иначе любой attacker отправит вам произвольный callback.
  • Чтобы не остаться без газа: subscription для VRF / Automation / Functions должна иметь автоматическое пополнение или мониторинг баланса LINK.

16. Песочница · Chainlink Converter в Remix

Iframe ниже открывает Converter.sol из Задачи 13.1 в Remix. В Remix VM нет реальных Chainlink Data Feed'ов, поэтому в задаче нужно подложить MockV3Aggregator (он в том же файле ниже) — это даёт детерминированные сценарии для тестов «stale price», «negative answer», «zero updatedAt» без поднятия Anvil-форка. 

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

interface AggregatorV3Interface {
    function decimals() external view returns (uint8);
    function latestRoundData() external view returns (
        uint80 roundId,
        int256 answer,
        uint256 startedAt,
        uint256 updatedAt,
        uint80 answeredInRound
    );
}

contract Converter {
    error StalePrice(uint256 updatedAt, uint256 nowTs);
    error NegativePrice(int256 answer);
    error StaleRound();

    AggregatorV3Interface public immutable feed;
    uint256 public immutable maxStaleSeconds;

    constructor(address _feed, uint256 _maxStaleSeconds) {
        feed = AggregatorV3Interface(_feed);
        maxStaleSeconds = _maxStaleSeconds;
    }

    /// Конвертирует amountWei (1e18 = 1 ETH) в USD * 1e18.
    function convertEthToUsd(uint256 amountWei) external view returns (uint256) {
        (uint80 roundId, int256 answer, , uint256 updatedAt, uint80 answeredInRound)
            = feed.latestRoundData();
        if (answer <= 0) revert NegativePrice(answer);
        if (updatedAt == 0 || block.timestamp - updatedAt > maxStaleSeconds) {
            revert StalePrice(updatedAt, block.timestamp);
        }
        if (answeredInRound < roundId) revert StaleRound();

        // ETH/USD feed имеет 8 decimals. Приводим к 1e18:
        uint256 price = uint256(answer) * 1e10;
        return amountWei * price / 1e18;
    }
}

/// @notice Мок для тестов в Remix VM (нет реального Chainlink на VM).
contract MockV3Aggregator is AggregatorV3Interface {
    uint8   private _decimals;
    int256  public  answer;
    uint256 public  updatedAt;
    uint80  public  roundId;

    constructor(uint8 d, int256 initial) {
        _decimals = d;
        answer = initial;
        updatedAt = block.timestamp;
        roundId = 1;
    }

    function decimals() external view returns (uint8) { return _decimals; }
    function setAnswer(int256 a) external { answer = a; updatedAt = block.timestamp; roundId++; }
    function setUpdatedAt(uint256 t) external { updatedAt = t; }
    function latestRoundData() external view returns (uint80, int256, uint256, uint256, uint80) {
        return (roundId, answer, updatedAt, updatedAt, roundId);
    }
}
