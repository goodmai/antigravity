Akash SDL (Stack Definition Language)

---
version: "2.0"
services:
  web:
    image: ghcr.io/me/app@sha256:abc...       # digest, не tag — критично!
    expose:
      - port: 8080
        as: 80
        to: [{ global: true }]
profiles:
  compute:
    web:
      resources:
        cpu: { units: 1 }
        memory: { size: 512Mi }
        storage: { size: 1Gi }
  placement:
    dcloud:
      attributes: { host: akash }
      signedBy: { anyOf: ["akash1..."] }      # доверяем подписанным аудиторам
      pricing:
        web: { denom: uakt, amount: 1000 }
deployment:
  web:
    dcloud:
      profile: web
      count: 1

# Создаём deployment, ждём bids, выбираем provider, lease
akash tx deployment create deploy.yml --from $WALLET --gas auto
akash query market bid list --owner $OWNER
akash tx market lease create --provider $PROVIDER ...
# Отправляем манифест выбранному провайдеру
akash provider send-manifest deploy.yml --provider $PROVIDER --from $WALLET

Артефакты на IPFS/Filecoin

# IPFS Kubo
ipfs add -r ./dist                  # → CID Qm...
ipfs name publish /ipfs/Qm...        # IPNS-указатель (как DNS)

# или web3.storage (управляемый Filecoin pinning)
w3 up ./dist                         # автоматический pinning

CID в адресе гарантирует content integrity — IPFS-имя это хеш контента.

Подмена невозможна (изменится сам адрес).

Использование в проде:

  • SBOM/SLSA-провенанс храним в Filecoin (immutable archive).
  • Static-сайты раздаём через IPFS-gateway (https://cf-ipfs.com/ipfs/$CID)

+ кэшируем через Cloudflare.

  • Артефакты релиза дублируем ghcr.io (быстро) + ipfs:// (immutable).

Лаба 25 — публикация контейнера в Akash + артефакт SBOM в IPFS, верификация.

Кейс: SDET-инженер запускает Playwright E2E-тесты в Akash

Задача: параллельно запустить 10 Playwright-воркеров в Akash, собрать HTML-отчёт и опубликовать его в IPFS. Экономия по сравнению с GitHub-hosted runners при долгих E2E-прогонах — от 60% (нет посекундной тарификации, только lease-стоимость).

SDL-манифест: test-runner + nginx report server

---
version: "2.0"
services:
  test-runner:
    image: ghcr.io/myorg/playwright@sha256:abc...   # digest — обязательно!
    env:
      - BASE_URL=https://staging.example.com
      - WORKERS=10
    expose:
      - port: 9323    # Prometheus metrics endpoint
        to: [{ global: false }]   # метрики — только внутри deployment
  report:
    image: nginx:alpine
    expose:
      - port: 80
        as: 80
        to: [{ global: true }]    # публичный доступ к HTML-отчёту
profiles:
  compute:
    test-runner:
      resources:
        cpu: { units: 4 }
        memory: { size: 4Gi }
        storage: { size: 5Gi }
    report:
      resources:
        cpu: { units: 0.5 }
        memory: { size: 256Mi }
        storage: { size: 1Gi }
  placement:
    dcloud:
      attributes: { host: akash }
      signedBy: { anyOf: ["akash1..."] }
      pricing:
        test-runner: { denom: uakt, amount: 500 }
        report: { denom: uakt, amount: 100 }
deployment:
  test-runner:
    dcloud:
      profile: test-runner
      count: 1
  report:
    dcloud:
      profile: report
      count: 1

GH Actions: публикация test report в IPFS

- name: Upload Playwright report to IPFS
  id: ipfs-upload
  run: |
    # Устанавливаем w3 CLI (web3.storage)
    npm install -g @web3-storage/w3cli
    # Авторизуемся через токен из секрета
    w3 login --no-browser --token "$W3_TOKEN"
    # Загружаем директорию с отчётом, получаем CID
    CID=$(w3 up ./playwright-report --no-wrap --json | jq -r '.root."/"."/"')
    echo "Report CID: $CID"
    # Записываем CID в GitHub Step Summary
    echo "## Playwright Test Report" >> $GITHUB_STEP_SUMMARY
    echo "IPFS CID: \`$CID\`" >> $GITHUB_STEP_SUMMARY
    echo "Gateway URL: https://cf-ipfs.com/ipfs/$CID" >> $GITHUB_STEP_SUMMARY
    echo "cid=$CID" >> $GITHUB_OUTPUT
  env:
    W3_TOKEN: ${{ secrets.W3_STORAGE_TOKEN }}
