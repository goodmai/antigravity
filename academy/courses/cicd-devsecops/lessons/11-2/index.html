Patroni (auto-failover для Postgres)

# patroni.yml
scope: cluster1
namespace: /service/
restapi: { listen: 0.0.0.0:8008 }
etcd: { hosts: etcd1:2379,etcd2:2379,etcd3:2379 }
bootstrap:
  dcs:
    ttl: 30
    loop_wait: 10
    retry_timeout: 10
    maximum_lag_on_failover: 1048576
    postgresql:
      use_pg_rewind: true
      parameters:
        wal_level: logical
        max_wal_senders: 10
        max_replication_slots: 10
        hot_standby: "on"
